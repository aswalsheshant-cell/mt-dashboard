// =====================================================================
// FUNCTION: fnCombineFolder
// Combines every .csv / .xlsx / .xlsb file found in a sub-folder into one
// table. Works for the "drop a file each month and click Refresh" workflow:
// add a new file to the folder, refresh, and the rows are appended.
//
// - Files whose name starts with "_" are ignored (so _TEMPLATE_*.csv and
//   _archive files never get loaded).
// - Adds [Data Source File] and [Refresh Date] audit columns automatically.
// - The header row is configurable; default is row 0.
//
// Usage in another query:
//   = fnCombineFolder(pRootFolder & "\RawDataFolders\Offtake_Monthly", 1)
// =====================================================================
let
    fnCombineFolder = (FolderPath as text, optional HeaderRow as nullable number) as table =>
        let
            Source = Folder.Files(FolderPath),
            HeaderIndex = if HeaderRow = null then 0 else Number.RoundDown(Number.From(HeaderRow)),

            // keep only real data files, skip templates / hidden / archive
            OnlyData = Table.SelectRows(
                Source,
                each ([Attributes]?[Hidden]? <> true)
                    and (Text.StartsWith([Name], "_") = false)
                    and (
                        Text.EndsWith(Text.Lower([Extension]), ".csv")
                        or Text.EndsWith(Text.Lower([Extension]), ".xlsx")
                        or Text.EndsWith(Text.Lower([Extension]), ".xlsb")
                    )
            ),

            // parse each file into a table depending on its type
            AddParsed = Table.AddColumn(OnlyData, "Parsed", each
                let
                    ext = Text.Lower([Extension]),
                    bin = [Content],
                    asTable =
                        if ext = ".csv" then
                            Csv.Document(bin, [Delimiter=",", Encoding=65001, QuoteStyle=QuoteStyle.Csv])
                        else
                            // xlsx / xlsb : take the first sheet
                            let
                                wb = Excel.Workbook(bin, null, true),
                                firstSheet = wb{0}[Data]
                            in firstSheet,
                    promoted =
                        if HeaderIndex = 0 then
                            Table.PromoteHeaders(asTable, [PromoteAllScalars=true])
                        else
                            let
                                skipped = Table.Skip(asTable, HeaderIndex),
                                promotedHeaders = Table.PromoteHeaders(skipped, [PromoteAllScalars=true])
                            in
                                promotedHeaders
                in
                    promoted
            ),

            // stamp source file + refresh timestamp on every row
            AddMeta = Table.AddColumn(AddParsed, "WithMeta", each
                Table.AddColumn(
                    Table.AddColumn([Parsed], "Data Source File", (r) => [Name], type text),
                    "Refresh Date", (r) => DateTime.LocalNow(), type datetime
                )
            ),

            Combined =
                if Table.RowCount(AddMeta) = 0 then
                    #table({"Data Source File","Refresh Date"}, {})
                else
                    Table.Combine(AddMeta[WithMeta])
        in
            Combined
in
    fnCombineFolder

// =====================================================================
// QUERY: Fact P&L  (built, not hardcoded)
// P&L is DERIVED at refresh time from Offtake NSV + the Assumption Table.
// If a future raw P&L file with actual cost/spend exists, drop it into a
// RawDataFolders\PnL_Monthly folder and union it here; otherwise the model

// that the P&L DAX measures sit on top of. No spend is hardcoded here.

    Source = #"Fact Offtake Sales",
    Grouped = Table.Group(
        Source,
        {"MonthStart", "Month", "FY Year", "Chain", "Brand", "Category"},
        {
            {"Offtake NSV", each List.Sum([Offtake NSV]), type number},
            {"Offtake Qty", each List.Sum([Offtake Qty]), type number},
            {"MRP Sales",   each List.Sum([MRP Sales]),   type number}
        }
    )
in
    Grouped

// =====================================================================
// QUERY: Fact Primary ShipTo   (grain = Month x Ship-To x Chain x Brand)
// Source = RawDataFolders\Primary_ShipTo_Monthly
//
// This is the ship-to-party primary feed where primary NSV is ALREADY
// allocated to Chain based on the secondary/offtake contribution % that
// month. (Direct ship-tos => one chain, Cont% = 100%. Distributor ship-tos
// => split across chains by Cont%.)
//
// The Cont% column is the dynamic, month-wise allocation driver (the
// "separate mapping table"). It is NOT hardcoded - it arrives in the monthly
// file and is also projected out as 'Primary Allocation Map' (query 34).
//
// Cleaning here is reproducible every month: brand/chain spellings are
// normalised and NSV (received in Lacs) is converted to absolute rupees so
// it lines up with Fact Offtake Sales.
// =====================================================================
let
    Source = fnCombineFolder(pRootFolder & "\RawDataFolders\Primary_ShipTo_Monthly"),

    Typed = Table.TransformColumnTypes(Source, {
        {"Month", type text},
        {"MonthStart", type date},
        {"FY Year", type text},
        {"Ship To Name", type text},
        {"Direct/Distributor", type text},
        {"Chain", type text},
        {"Zone", type text},
        {"State", type text},
        {"Brand", type text},
        {"Primary NSV", type number},
        {"MRP Value", type number},
        {"Cont%", type number}
    }),

    Trimmed = Table.TransformColumns(Typed, {
        {"Ship To Name", each Text.Trim(Text.From(_)), type text},
        {"Chain", each Text.Trim(Text.From(_)), type text},
        {"Brand", each Text.Trim(Text.From(_)), type text}
    }),

    // --- normalise brand spellings (keep in sync with Brand Master) ---
    FixBrand = Table.ReplaceValue(Trimmed, "BBLUNT", "BBlunt", Replacer.ReplaceText, {"Brand"}),

    // --- normalise chain spellings (keep in sync with Chain Master) ---
    FixChain1 = Table.ReplaceValue(FixBrand, "Vmm",       "Vishal Mega Mart", Replacer.ReplaceValue, {"Chain"}),
    FixChain2 = Table.ReplaceValue(FixChain1,"VMM",       "Vishal Mega Mart", Replacer.ReplaceValue, {"Chain"}),
    FixChain3 = Table.ReplaceValue(FixChain2,"ARAMBAGH",  "Arambagh",         Replacer.ReplaceValue, {"Chain"}),
    FixChain4 = Table.ReplaceValue(FixChain3,"Reliance",  "Reliance Retail",  Replacer.ReplaceValue, {"Chain"}),
    FixChain5 = Table.ReplaceValue(FixChain4,"Dmart",     "D-Mart",           Replacer.ReplaceValue, {"Chain"}),
    FixChain6 = Table.ReplaceValue(FixChain5,"Spencer",   "Spencers",         Replacer.ReplaceValue, {"Chain"}),
    FixChain7 = Table.ReplaceValue(FixChain6,"H&G",       "Health & Glow",    Replacer.ReplaceValue, {"Chain"}),

    // If the file ever arrives in Lacs again instead of absolute rupees,
    // flip this flag. The seed already stores absolute rupees, so default = false.
    ConvertFromLacs = false,
    Scaled = if ConvertFromLacs
        then Table.TransformColumns(FixChain7, {
                {"Primary NSV", each _ * 100000, type number},
                {"MRP Value",  each _ * 100000, type number}})
        else FixChain7,

    // Direct rows may have blank Cont% -> treat as 100%
    FillCont = Table.ReplaceValue(Scaled, null, 1.0, Replacer.ReplaceValue, {"Cont%"}),

    // rebuild MonthStart if the source only had the text label
    AddMonthStart = Table.AddColumn(FillCont, "MonthStartCalc", each
        if [MonthStart] <> null then [MonthStart]
        else
            let m = Text.Start(Text.BeforeDelimiter([Month], "'"), 3),
                yy = Text.AfterDelimiter([Month], "'"),
                mn = Record.FieldOrDefault([Jan=1,Feb=2,Mar=3,Apr=4,May=5,Jun=6,Jul=7,Aug=8,Sep=9,Oct=10,Nov=11,Dec=12], m, null)
            in if mn = null then null else #date(2000 + Number.FromText(yy), mn, 1),
        type date),

    Cleaned = Table.SelectRows(AddMonthStart, each [Primary NSV] <> null)
in
    Cleaned

    // a single digit-counting heuristic: "FY26" (short) names the FY ending
    // in 2026, and so does "FY25-26" (span, named by its END year).
    //   span:  first "dd-dd" anywhere, second = first + 1 -> "FY" & second
    //          ("FY25-26", "FY'25-26", "FY_25-26", "2025-26")
    //   short: whole text (spaces removed) is "FY" + optional 0 + 2 digits
    //          ("FY26", "FY 26", "FY026")

    // tests/powerbi/pq39_fy_parser_cases.pq runs this exact function over
    // tests/powerbi/fy_parser_cases.csv in Desktop (text kept identical by
    // tests/test_pq39_fy_parser.py).

        let
            _s = if fy = null then "" else Text.Remove(Text.Upper(Text.Trim(Text.From(fy))), {" "}),
            _isDD = (t as text) as logical =>
                Text.Length(t) = 2 and List.AllTrue(List.Transform(Text.ToList(t), each Text.Contains("0123456789", _))),
            _spans = List.Select(Text.PositionOf(_s, "-", Occurrence.All), each

            _a = if _p = null then null else Number.FromText(Text.Middle(_s, _p - 2, 2)),

            _span = if _p <> null and _b = Number.Mod(_a + 1, 100) then "FY" & Text.PadStart(Text.From(_b), 2, "0") else null,
            _body = if Text.StartsWith(_s, "FY") then Text.Middle(_s, 2) else null,
            _digits = if _body <> null and Text.Length(_body) = 3 and Text.StartsWith(_body, "0") then Text.Middle(_body, 1) else _body,
            _short = if _digits <> null and _isDD(_digits) then "FY" & _digits else null
        in
            if _span <> null then _span else _short,

             Oct=10, October=10, Nov=11, November=11, Dec=12, December=12,
             Jan=1, January=1, Feb=2, February=2, Mar=3, March=3],
            Text.Proper(Text.Trim(Text.From([Month]))), null),
        Int64.Type),
    AddMonthStart = Table.AddColumn(AddMonthNum, "MonthStart", each
        if [_FyStartYear] = null or [_MonthNum] = null then null

        type date),
    RemoveHelpers = Table.RemoveColumns(AddMonthStart, {"_FyTag", "_FyStartYear", "_MonthNum"}),

    // The seed file ships 3 template rows marked "EXAMPLE ROW" in Remarks to show

// =====================================================================
// QUERY: Secondary Sales Efficiency  (pre-aggregated helper table)
// Grain: FY Year x Month x Chain x Brand
// Computes sell-through (offtake / primary) at chain x brand x month level,
// returning the core efficiency metrics for the Distribution tab's
// "Offtake Efficiency" sub-section and the Chain-wise Performance page.
//
// Columns:
//   FY Year, Month, MonthStart, Chain, Brand
//   Primary NSV      — Lakh, from Fact Primary Article (DIST-allocated)
//   Offtake NSV      — Lakh, from Fact Offtake Sales
//   Sell-Through Pct — Offtake NSV / Primary NSV (null-safe, capped at 200%
//                      to exclude months where primary = 0 but offtake > 0
//                      due to pipeline stock; un-capped rows surfaced in
//                      Sell-Through Outlier flag)
//   Coverage Gap Pct — 1 − Sell-Through (positive = under-liquidation)
//   Inventory Signal  — "Over-stocked" / "On-track" / "Under-liquidation"
//                      thresholds: ≥ 120% over-stocked, 80–120% on-track,
//                      < 80% under-liquidation (adjust in Assumption Table)
//   Outlier Flag     — true when Primary NSV = 0 and Offtake NSV > 0
//                      (pipeline / opening-stock liquidation — exclude from
//                      efficiency KPIs, review in Distribution Gap page)
//
// Join: this query aggregates both facts separately then merges — it does NOT
// require a shared row-level key between primary and offtake. Chain and Brand
// are the natural join keys; "Reliance Retail" / "Reliance Retail Limited"
// divergence is resolved by the canonical Chain names established in
// Fact Primary Article ([Chain]) and Fact Offtake Sales ([Chain]).
// =====================================================================
let
    // ---- 1. Aggregate Primary to Chain x Brand x Month ----
    PrimSrc = Table.SelectColumns(
        #"Fact Primary Article",
        {"FY Year", "Month", "MonthStart", "Chain", "Brand", "Primary NSV"}),

    PrimAgg = Table.Group(PrimSrc,
        {"FY Year", "Month", "MonthStart", "Chain", "Brand"},
        {{"Primary NSV", each List.Sum([Primary NSV]), type number}}),

    // ---- 2. Aggregate Offtake to Chain x Brand x Month ----
    OftSrc = Table.SelectColumns(
        #"Fact Offtake Sales",
        {"FY Year", "Month", "MonthStart", "Chain", "Brand", "Offtake NSV"}),

    OftAgg = Table.Group(OftSrc,
        {"FY Year", "Month", "MonthStart", "Chain", "Brand"},
        {{"Offtake NSV", each List.Sum([Offtake NSV]), type number}}),

    // ---- 3. Full outer join (keep rows present in either fact) ----
    Joined = Table.NestedJoin(
        PrimAgg, {"FY Year", "Month", "MonthStart", "Chain", "Brand"},
        OftAgg,  {"FY Year", "Month", "MonthStart", "Chain", "Brand"},
        "Oft", JoinKind.FullOuter),

    Expanded = Table.ExpandTableColumn(Joined, "Oft", {"Offtake NSV"}),

    FillPrim = Table.ReplaceValue(Expanded, null, 0, Replacer.ReplaceValue, {"Primary NSV"}),
    FillOft  = Table.ReplaceValue(FillPrim,  null, 0, Replacer.ReplaceValue, {"Offtake NSV"}),

    // ---- 4. Outlier flag: offtake from pipeline stock (Primary = 0) ----
    AddOutlier = Table.AddColumn(FillOft, "Outlier Flag", each
        [Primary NSV] = 0 and [Offtake NSV] > 0, type logical),

    // ---- 5. Sell-Through Pct (null when Primary=0 to avoid ÷0 distortion) ----
    AddST = Table.AddColumn(AddOutlier, "Sell-Through Pct", each
        if [Primary NSV] = 0 then null
        else Number.Round([Offtake NSV] / [Primary NSV] * 100, 1),
        type number),

    // ---- 6. Raw (uncapped) for outlier review ----
    AddSTRaw = Table.AddColumn(AddST, "Sell-Through Raw", each
        if [Primary NSV] = 0 then null
        else [Offtake NSV] / [Primary NSV] * 100,
        type number),

    // ---- 7. Capped Sell-Through (≤ 200%) for efficiency KPI visuals ----
    AddSTCapped = Table.AddColumn(AddSTRaw, "Sell-Through Capped", each
        if [Sell-Through Raw] = null then null
        else Number.Min([Sell-Through Raw], 200),
        type number),

    // ---- 8. Coverage Gap ----
    AddGap = Table.AddColumn(AddSTCapped, "Coverage Gap Pct", each
        if [Sell-Through Pct] = null then null
        else Number.Round(100 - [Sell-Through Pct], 1),
        type number),

    // ---- 9. Inventory Signal (thresholds sourced from Assumption Table) ----
    //   Over-stocked      : ST ≥ 120%  (high offtake vs. primary = inventory clearing)
    //   On-track          : 80% ≤ ST < 120%
    //   Under-liquidation : ST < 80%   (slow sell-through = excess stock risk)
    //   No Primary        : Primary = 0 (outlier/pipeline — review separately)
    AddSignal = Table.AddColumn(AddGap, "Inventory Signal", each
        if [Outlier Flag]               then "No Primary Data"
        else if [Sell-Through Pct] = null then "No Primary Data"
        else if [Sell-Through Pct] >= 120 then "Over-stocked"
        else if [Sell-Through Pct] >= 80  then "On-track"
        else                                   "Under-liquidation",
        type text),

    // ---- 10. Signal Sort (for consistent legend ordering) ----
    AddSignalSort = Table.AddColumn(AddSignal, "Signal Sort", each
        Record.FieldOrDefault(
            [#"On-track"=1, #"Under-liquidation"=2, #"Over-stocked"=3, #"No Primary Data"=4],
            [Inventory Signal], 5),
        type number),

    // ---- 11. Final select and type ----
    Final = Table.SelectColumns(AddSignalSort, {
        "FY Year", "Month", "MonthStart", "Chain", "Brand",
        "Primary NSV", "Offtake NSV",
        "Sell-Through Pct", "Sell-Through Raw", "Sell-Through Capped",
        "Coverage Gap Pct", "Inventory Signal", "Signal Sort", "Outlier Flag"
    }),

    Typed = Table.TransformColumnTypes(Final, {
        {"FY Year", type text}, {"Month", type text}, {"MonthStart", type date},
        {"Chain", type text}, {"Brand", type text},
        {"Primary NSV", type number}, {"Offtake NSV", type number},
        {"Sell-Through Pct", type number}, {"Sell-Through Raw", type number},
        {"Sell-Through Capped", type number}, {"Coverage Gap Pct", type number},
        {"Inventory Signal", type text}, {"Signal Sort", Int64.Type},
        {"Outlier Flag", type logical}
    })
in
    Typed

// =====================================================================
// QUERY: Fact Claim Master  (Chain-level Q1 FY27 expense)
// Grain: Quarter × Chain × Expense Category
//
// Source: PowerBI/RawDataFolders/ClaimMaster_Quarterly/
//         Files: claim_master_chain_AprJun_2026.csv  (and future quarters)
//
// Required columns:
//   Period           — e.g. "Apr-Jun 2026"
//   FY_Year          — FY27
//   Quarter          — Q1
//   Chain            — canonical chain name
//   Chain_Promo_Lakh — on-invoice promo passed to chain
//   Rate_Diff_Lakh   — extra margin / rate difference
//   Freight_Lakh     — freight / transportation
//   Incentive_Lakh   — incentive
//   Off_Invoice_Lakh — off-invoice / debit note promo
//   Visibility_Lakh  — visibility spend
//   Total_Claim_Lakh — sum of all above
//
// Companion queries (same folder, same refresh cycle):
//   claim_master_brand_AprJun_2026.csv    → loaded in 45b_ClaimBrand
//   claim_master_distributor_AprJun_2026.csv → loaded in 45c_ClaimDistributor
//
// Refresh: drop a new quarterly CSV into ClaimMaster_Quarterly/ → Refresh.
// =====================================================================
let
    Source = fnCombineFolder(
        pRootFolder & "\PowerBI\RawDataFolders\ClaimMaster_Quarterly",
        {"Period", "FY_Year", "Quarter", "Chain", "Source_Chain",
         "Chain_Promo_Lakh", "Rate_Diff_Lakh", "Freight_Lakh",
         "Incentive_Lakh", "Off_Invoice_Lakh", "Visibility_Lakh",
         "Total_Claim_Lakh"}
    ),

    // Filter to chain-level file pattern
    ChainOnly = Table.SelectRows(Source, each
        Table.RowCount(Table.SelectRows(Source, each [Chain] <> null)) > 0),

    Typed = Table.TransformColumnTypes(Source, {
        {"Period",            type text},
        {"FY_Year",           type text},
        {"Quarter",           type text},
        {"Chain",             type text},
        {"Source_Chain",      type text},
        {"Chain_Promo_Lakh",  type number},
        {"Rate_Diff_Lakh",    type number},
        {"Freight_Lakh",      type number},
        {"Incentive_Lakh",    type number},
        {"Off_Invoice_Lakh",  type number},
        {"Visibility_Lakh",   type number},
        {"Total_Claim_Lakh",  type number}
    }),

    // ── Unpivot to long form: Chain × Expense_Category × Amount ────
    Unpivoted = Table.UnpivotOtherColumns(Typed,
        {"Period", "FY_Year", "Quarter", "Chain", "Source_Chain"},
        "Expense_Category_Raw", "Amount_Lakh"),

    // ── Clean category names (strip "_Lakh" suffix) ─────────────────
    WithCatClean = Table.AddColumn(Unpivoted, "Expense_Category", each
        let raw = [Expense_Category_Raw]
        in  if raw = "Chain_Promo_Lakh"  then "Chain Promo (On Invoice)"
            else if raw = "Rate_Diff_Lakh"   then "Extra Margin / Rate Difference"
            else if raw = "Freight_Lakh"     then "Freight / Transportation"
            else if raw = "Incentive_Lakh"   then "Incentive"
            else if raw = "Off_Invoice_Lakh" then "Off Invoice / Debit Note Promo"
            else if raw = "Visibility_Lakh"  then "Visibility"
            else raw,
        type text),

    // ── Exclude zero rows (cleaner model) ──────────────────────────
    NonZero = Table.SelectRows(WithCatClean, each [Amount_Lakh] <> 0),

    Final = Table.SelectColumns(NonZero, {
        "Period", "FY_Year", "Quarter",
        "Chain", "Source_Chain",
        "Expense_Category", "Amount_Lakh"
    })
in
    Final
