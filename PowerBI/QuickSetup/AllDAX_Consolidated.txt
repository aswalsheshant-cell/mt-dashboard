# STEP 01/14  --  SOURCE FILE: 00_DateTable.dax
# TARGET: Date Table (calculated TABLE, not a measure)
# NOTE: Modeling > New table > paste the WHOLE file as one calculated table. Then mark as date table on [Date]. NOT part of the _Measures bulk-paste below.
##############################################################################
// =====================================================================
// DATE TABLE  —  create via:  Modeling > New table  (paste this DAX)
// Marked as the model's date table (Table tools > Mark as date table > [Date]).
// Covers FY Apr->Mar (Indian financial year). Spans 3 yrs back to next FY so
// YoY and forecast months always resolve. Adjust the start/end if needed.
// =====================================================================
Date Table =
VAR _Start = DATE(2024, 4, 1)        // first day of earliest FY you report
// S1: derive the end from the data (+12 months headroom for YoY/forecast) so
// FY29+ appear automatically as months arrive — no annual hand-edit. Falls back
// to Mar-2028 if the fact is empty on first load.
VAR _End   = EOMONTH ( COALESCE ( MAX ( 'Fact Offtake Sales'[MonthStart] ), DATE(2028, 3, 31) ), 12 )
RETURN
ADDCOLUMNS(
    CALENDAR(_Start, _End),
    "MonthStart",      EOMONTH([Date], -1) + 1,
    "Year",            YEAR([Date]),
    "Month No",        MONTH([Date]),
    "Month Name",      FORMAT([Date], "mmm"),
    // Indian FY: Apr=1 ... Mar=12
    "FY Month No",     IF(MONTH([Date]) >= 4, MONTH([Date]) - 3, MONTH([Date]) + 9),
    "FY Year",
        VAR y = YEAR([Date])
        RETURN IF(MONTH([Date]) >= 4,
                 FORMAT(y, "00") & "-" & FORMAT(y + 1 - 2000, "00"),
                 FORMAT(y - 1, "00") & "-" & FORMAT(y - 2000, "00")),
    "Quarter",
        VAR fm = IF(MONTH([Date]) >= 4, MONTH([Date]) - 3, MONTH([Date]) + 9)
        RETURN "Q" & ROUNDUP(fm / 3, 0),
    // label used everywhere in the deck, e.g. May'26
    "Month",           FORMAT([Date], "mmm") & "'" & FORMAT([Date], "yy"),
    "Month Year Sort", YEAR([Date]) * 100 + MONTH([Date]),
    "Is Future",       IF([Date] > TODAY(), TRUE(), FALSE())
)

// -----------------------------------------------------------------
// AFTER creating: set Sort-by columns
//   [Month]      sort by [Month Year Sort]
//   [Month Name] sort by [Month No]
//   [Quarter]    sort by [FY Month No]
// Relationships (single direction, 1 -> *):
//   Date Table[MonthStart] -> Fact Offtake Sales[MonthStart]
//   Date Table[Date]       -> Fact Primary Sales[Week Start Date]
//   Date Table[MonthStart] -> Fact P&L[MonthStart]
//   Date Table[MonthStart] -> Fact Nielsen[MonthStart]
//   Date Table[MonthStart] -> Fact TDP[MonthStart]
//   Date Table[MonthStart] -> Targets[MonthStart]
// -----------------------------------------------------------------

# STEP 02/14  --  SOURCE FILE: 01_CoreMeasures.dax
# TARGET: _Measures table
##############################################################################
// =====================================================================
// CORE MEASURES  —  create a "_Measures" table (Enter Data, one blank col)
// and add each measure below to it. Every division uses DIVIDE().
// Convention: "NSV" headline = Offtake NSV (matches the leadership deck).
// =====================================================================

// ---------- BASE TOTALS ----------
Total Offtake NSV = SUM ( 'Fact Offtake Sales'[Offtake NSV] )
Total Offtake Qty = SUM ( 'Fact Offtake Sales'[Offtake Qty] )
Total Primary NSV = SUM ( 'Fact Primary Sales'[Primary NSV] )
Total Primary Qty = SUM ( 'Fact Primary Sales'[Primary Qty] )
Total MRP Sales   = SUM ( 'Fact Offtake Sales'[MRP Sales] )

// Headline NSV used by most visuals (Offtake basis). Swap to Primary if needed.
NSV = [Total Offtake NSV]

// ---------- INDIAN FORMAT HELPERS (display measures) ----------
NSV (Cr) = DIVIDE ( [NSV], 10000000 )
NSV (Lacs) = DIVIDE ( [NSV], 100000 )
// Smart label: Cr if >= 1 Cr else Lacs
NSV Label =
VAR v = [NSV]
RETURN
    IF (
        ABS ( v ) >= 10000000,
        FORMAT ( DIVIDE ( v, 10000000 ), "0.00" ) & " Cr",
        FORMAT ( DIVIDE ( v, 100000 ), "0.0" ) & " L"
    )

// ---------- TIME INTELLIGENCE (Offtake basis) ----------
// S2: renamed from the misleading "Latest Month Sales" — this is a CUMULATIVE
// running total up to the latest in-context date, NOT a single month. Use
// [Latest Month NSV] below for the single-month KPI.
NSV Cumulative To Latest =
VAR _lastDate = MAX ( 'Date Table'[Date] )
RETURN CALCULATE ( [NSV], 'Date Table'[Date] <= _lastDate )
// In a month-grain visual [NSV] already = the selected month; the explicit
// "latest month" KPI for cards:
Latest Month NSV =
VAR _lm = CALCULATE ( MAX ( 'Fact Offtake Sales'[MonthStart] ), ALL ( 'Date Table' ) )
RETURN CALCULATE ( [NSV], 'Date Table'[MonthStart] = _lm )

Previous Month NSV =
CALCULATE ( [NSV], DATEADD ( 'Date Table'[Date], -1, MONTH ) )

MoM Growth % =
VAR cur = [NSV]
VAR prev = [Previous Month NSV]
RETURN DIVIDE ( cur - prev, prev )

Last Year Same Month NSV =
CALCULATE ( [NSV], DATEADD ( 'Date Table'[Date], -12, MONTH ) )

YoY Growth % =
DIVIDE ( [NSV] - [Last Year Same Month NSV], [Last Year Same Month NSV] )

// ---------- ROLLING AVERAGES ----------
L3M Average Sales =
VAR _curMonth = MAX ( 'Date Table'[MonthStart] )
RETURN
    CALCULATE (
        DIVIDE ( [NSV], 3 ),
        DATESINPERIOD ( 'Date Table'[Date], EOMONTH ( _curMonth, -1 ), -3, MONTH )
    )

L6M Average Sales =
VAR _curMonth = MAX ( 'Date Table'[MonthStart] )
RETURN
    CALCULATE (
        DIVIDE ( [NSV], 6 ),
        DATESINPERIOD ( 'Date Table'[Date], EOMONTH ( _curMonth, -1 ), -6, MONTH )
    )

Growth vs L3M % = DIVIDE ( [NSV] - [L3M Average Sales], [L3M Average Sales] )

// ---------- CONTRIBUTION & RANK ----------
Contribution % =
DIVIDE (
    [NSV],
    CALCULATE ( [NSV], ALLSELECTED ( 'Chain Master'[Chain] ), ALLSELECTED ( 'Brand Master'[Brand] ),
        ALLSELECTED ( 'Category Master'[Category] ) )
)

Rank by Sales =
IF ( ISINSCOPE ( 'Chain Master'[Chain] ),
    RANKX ( ALLSELECTED ( 'Chain Master'[Chain] ), [NSV],, DESC ),
    RANKX ( ALLSELECTED ( 'Brand Master'[Brand] ), [NSV],, DESC )
)

// ---------- PRIMARY vs OFFTAKE ----------
Primary vs Offtake Gap = [Total Primary NSV] - [Total Offtake NSV]
Primary vs Offtake Gap % = DIVIDE ( [Total Primary NSV] - [Total Offtake NSV], [Total Offtake NSV] )

// ---------- GROWTH ARROW (for tables / cards) ----------
MoM Arrow =
VAR g = [MoM Growth %]
RETURN IF ( ISBLANK ( g ), "", IF ( g >= 0, "▲ ", "▼ " ) ) & FORMAT ( g, "0.0%" )

YoY Arrow =
VAR g = [YoY Growth %]
RETURN IF ( ISBLANK ( g ), "", IF ( g >= 0, "▲ ", "▼ " ) ) & FORMAT ( g, "0.0%" )

// Conditional-format colour (use in "Font color > fx > Field value")
Growth Colour =
SWITCH ( TRUE (),
    [MoM Growth %] > 0, "#2D9B7F",   // teal/green positive
    [MoM Growth %] < 0, "#C0392B",   // red negative
    "#7F8C8D"                        // grey neutral
)

// ---------- TOP / BOTTOM MOVERS (Executive Summary) ----------
Top Growing Chain =
VAR t = TOPN ( 1, ALLSELECTED ( 'Chain Master'[Chain] ), [MoM Growth %], DESC )
RETURN CONCATENATEX ( t, 'Chain Master'[Chain], ", " )

Top Declining Chain =
VAR t = TOPN ( 1, ALLSELECTED ( 'Chain Master'[Chain] ), [MoM Growth %], ASC )
RETURN CONCATENATEX ( t, 'Chain Master'[Chain], ", " )

Top Growing Brand =
VAR t = TOPN ( 1, ALLSELECTED ( 'Brand Master'[Brand] ), [MoM Growth %], DESC )
RETURN CONCATENATEX ( t, 'Brand Master'[Brand], ", " )

Top Growing Category =
VAR t = TOPN ( 1, ALLSELECTED ( 'Category Master'[Category] ), [MoM Growth %], DESC )
RETURN CONCATENATEX ( t, 'Category Master'[Category], ", " )

# TARGET: _Measures table

// =====================================================================
// P&L MEASURES  (actuals-first, else Assumption Table — never hardcoded)
// Logic: if a raw P&L actual exists use it; otherwise resolve the most
// specific matching row in 'Assumption Table' (Month>Chain>Brand>Category),
// falling back to "ALL". All divisions use DIVIDE().
// =====================================================================

// ---- Helper: resolve an assumption value with ALL-fallback ----
// Returns the % / spend for the current Month x Chain x Brand x Category
// context, preferring the most specific row, else the ALL row.
_Assumption Gross Margin % =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
VAR _ch = SELECTEDVALUE ( 'Chain Master'[Chain], "ALL" )
VAR _br = SELECTEDVALUE ( 'Brand Master'[Brand], "ALL" )
VAR _cat = SELECTEDVALUE ( 'Category Master'[Category], "ALL" )
VAR _specific =
    CALCULATE ( MAX ( 'Assumption Table'[Gross Margin %] ),
        FILTER ( ALL ( 'Assumption Table' ),
            'Assumption Table'[Month] = _m
            && ( 'Assumption Table'[Chain] = _ch || 'Assumption Table'[Chain] = "ALL" )
            && ( 'Assumption Table'[Brand] = _br || 'Assumption Table'[Brand] = "ALL" )
            && ( 'Assumption Table'[Category] = _cat || 'Assumption Table'[Category] = "ALL" ) ) )
RETURN COALESCE ( _specific, 0.50 )

_Assumption Trade Spend % =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
COALESCE (
    CALCULATE ( MAX ( 'Assumption Table'[Trade Spend %] ),
        FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) ), 0 )

_Assumption Visibility Spend =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
COALESCE (
    CALCULATE ( SUM ( 'Assumption Table'[Visibility Spend] ),
        FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) ), 0 )

_Assumption Scheme Spend =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
COALESCE (
    CALCULATE ( SUM ( 'Assumption Table'[Scheme Spend] ),
        FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) ), 0 )

// ---- P&L line items ----
Net Sales = [NSV]                                   // NSV = Net Sales Value (offtake)
Gross Sales = COALESCE ( [Total MRP Sales], DIVIDE ( [NSV], 1 - 0.0 ) )  // use MRP Sales as gross proxy

Gross Margin = [NSV] * [_Assumption Gross Margin %]
Gross Margin % = DIVIDE ( [Gross Margin], [NSV] )

Trade Spend = [NSV] * [_Assumption Trade Spend %]
Visibility Spend = [_Assumption Visibility Spend]
Scheme Spend = [_Assumption Scheme Spend]
Total Spend = [Trade Spend] + [Visibility Spend] + [Scheme Spend]

Spend % of NSV = DIVIDE ( [Total Spend], [NSV] )

Contribution Margin = [Gross Margin] - [Total Spend]
CM % = DIVIDE ( [Contribution Margin], [NSV] )

// ---- ranking helper for the profitability page ----
Chain CM Rank =
RANKX ( ALLSELECTED ( 'Chain Master'[Chain] ), [Contribution Margin],, DESC )

// ---- "growth profitable or only sales-led?" flag ----
Profitability Flag =
VAR g = [MoM Growth %]
VAR cm = [CM %]
RETURN
    SWITCH ( TRUE (),
        g > 0 && cm >= 0.25, "Profitable Growth",
        g > 0 && cm < 0.25,  "Sales-led (watch margin)",
        g <= 0 && cm >= 0.25, "Margin-protected decline",
        "Needs attention"
    )

# STEP 04/14  --  SOURCE FILE: 03_Forecast_Measures.dax
# TARGET: _Measures table
##############################################################################
// =====================================================================
// FORECAST MEASURES  —  TY-TARGET DRIVEN  (replaces FY26 uplift logic)
// =====================================================================
// FY27 Forecast IS the TY target shared by the sales team (table 'Targets').
// The monthly target is allocated down to Chain / Zone / Brand / Category /
// Sales Person using each dimension's FY26 actual-offtake share, so EVERY
// breakdown still sums EXACTLY to the TY target total. Records with no
// mapping are NOT dropped — they bucket into "Unmapped".
//
// Allocation identity:  Σ(all dims) Forecast = TY Target  (variance = 0).
// =====================================================================

// ---------- Authoritative totals ----------
TY Target = CALCULATE ( SUM ( Targets[Target NSV] ), Targets[FY Year] = "26-27" )
TY Target (Cr) = DIVIDE ( [TY Target], 10000000 )

FY26 Actual Offtake = CALCULATE ( [Total Offtake NSV], 'Date Table'[FY Year] = "25-26" )
FY26 Actual Offtake (Cr) = DIVIDE ( [FY26 Actual Offtake], 10000000 )

// ---------- Allocation key = dimension share of FY26 offtake ----------
// Denominator is the FY26 offtake of the *whole* visible selection, so the
// shares across the axis sum to 1 and the target is fully distributed.
_FY26 Total Offtake (All Dims) =
CALCULATE (
    [FY26 Actual Offtake],
    ALLSELECTED ( 'Chain Master'[Chain] ),
    ALLSELECTED ( 'Brand Master'[Brand] ),
    ALLSELECTED ( 'Category Master'[Category] ),
    ALLSELECTED ( 'Zone State Master'[Zone] ),
    ALLSELECTED ( 'Sales Team Mapping'[Sales Person] )
)

FY26 Allocation Share = DIVIDE ( [FY26 Actual Offtake], [_FY26 Total Offtake (All Dims)] )

// ---------- FY27 Forecast (= allocated TY target) ----------
// At grand total the share = 1 so Forecast = TY Target exactly.
FY27 Forecast =
SUMX (
    VALUES ( 'Date Table'[MonthStart] ),
    VAR _monthTarget = [TY Target]
    VAR _share = [FY26 Allocation Share]
    RETURN _monthTarget * _share
)
FY27 Forecast (Cr) = DIVIDE ( [FY27 Forecast], 10000000 )

// Final Forecast keeps the old name used elsewhere, now target-driven.
Final Forecast = [FY27 Forecast]

// ---------- Unmapped bucket (do NOT exclude) ----------
// FY26 offtake whose Chain / Brand / Category is absent from the masters,
// expressed as a share, so the target portion it represents shows as Unmapped.
_FY26 Offtake Unmapped =
CALCULATE (
    [FY26 Actual Offtake],
    FILTER ( 'Fact Offtake Sales',
        NOT ( 'Fact Offtake Sales'[Chain]    IN VALUES ( 'Chain Master'[Chain] ) )
        || NOT ( 'Fact Offtake Sales'[Brand]    IN VALUES ( 'Brand Master'[Brand] ) )
        || NOT ( 'Fact Offtake Sales'[Category] IN VALUES ( 'Category Master'[Category] ) )
    )
)
Forecast - Unmapped =
[TY Target] * DIVIDE ( [_FY26 Offtake Unmapped], [_FY26 Total Offtake (All Dims)] )

// ---------- Sales-person ownership (via Sales Team Mapping x store offtake) ----------
// SO FY26 offtake = Σ over their stores of store offtake × Cont%.
SO FY26 Offtake =
SUMX (
    'Sales Team Mapping',
    VAR _store = 'Sales Team Mapping'[Store Code]
    VAR _w = 'Sales Team Mapping'[Cont%]
    RETURN _w * CALCULATE ( [FY26 Actual Offtake],
        'Fact Offtake Sales'[Store Code] = _store )
)
SO Forecast (Owned Target) =
[TY Target] * DIVIDE ( [SO FY26 Offtake],
    CALCULATE ( [SO FY26 Offtake], ALLSELECTED ( 'Sales Team Mapping'[Sales Person] ) ) )

// ---------- Gap vs FY26 & required growth ----------
Gap vs FY26 = [FY27 Forecast] - [FY26 Actual Offtake]
Required Growth % = DIVIDE ( [FY27 Forecast] - [FY26 Actual Offtake], [FY26 Actual Offtake] )

// ---------- Accuracy (once FY27 actuals start landing) ----------
Has Actual = IF ( [Total Offtake NSV] > 0, TRUE (), FALSE () )
Actual NSV = [Total Offtake NSV]
Forecast vs Actual = [Actual NSV] - [Final Forecast]
Forecast vs Actual % = DIVIDE ( [Actual NSV] - [Final Forecast], [Final Forecast] )
Forecast Accuracy % =
IF ( [Has Actual],
    1 - DIVIDE ( ABS ( [Actual NSV] - [Final Forecast] ), [Actual NSV] ),
    BLANK () )

// combined actual-or-forecast line for the trend chart
Actual or Forecast = IF ( [Has Actual], [Actual NSV], [Final Forecast] )

# TARGET: _Measures table

// =====================================================================
// NIELSEN MARKET SHARE MEASURES
// Source: 'Fact Nielsen'. Share can come pre-computed (Value Market Share %)
// or be derived from Our Brand Sales / Market Value Sales.
// =====================================================================

Market Value Sales = SUM ( 'Fact Nielsen'[Market Value Sales] )
Our Brand Sales    = SUM ( 'Fact Nielsen'[Our Brand Sales] )

// Prefer supplied share %, else derive
Market Share % =
VAR _supplied = AVERAGE ( 'Fact Nielsen'[Value Market Share %] )
RETURN COALESCE ( _supplied, DIVIDE ( [Our Brand Sales], [Market Value Sales] ) )

Market Share Volume % = AVERAGE ( 'Fact Nielsen'[Volume Market Share %] )

Prev Month Market Share % =
CALCULATE ( [Market Share %], DATEADD ( 'Date Table'[Date], -1, MONTH ) )

Last Year Market Share % =
CALCULATE ( [Market Share %], DATEADD ( 'Date Table'[Date], -12, MONTH ) )

Market Share MoM Change = [Market Share %] - [Prev Month Market Share %]
Market Share YoY Change = [Market Share %] - [Last Year Market Share %]

// basis points (1% = 100 bps)
Market Share BPS Change = [Market Share MoM Change] * 10000
Market Share YoY BPS    = [Market Share YoY Change] * 10000

MS BPS Label =
VAR b = [Market Share BPS Change]
RETURN IF ( ISBLANK ( b ), "", IF ( b >= 0, "▲ +", "▼ " ) ) & FORMAT ( b, "0" ) & " bps"

// category & competition movement
Category Value Growth % =
DIVIDE ( [Market Value Sales] - CALCULATE ( [Market Value Sales], DATEADD ( 'Date Table'[Date], -1, MONTH ) ),
         CALCULATE ( [Market Value Sales], DATEADD ( 'Date Table'[Date], -1, MONTH ) ) )

Our Brand Growth % =
DIVIDE ( [Our Brand Sales] - CALCULATE ( [Our Brand Sales], DATEADD ( 'Date Table'[Date], -1, MONTH ) ),
         CALCULATE ( [Our Brand Sales], DATEADD ( 'Date Table'[Date], -1, MONTH ) ) )

// "are we growing faster than the category?" (share gain check)
Share Gain Flag =
IF ( [Our Brand Growth %] > [Category Value Growth %], "Gaining Share", "Losing Share" )

# STEP 06/14  --  SOURCE FILE: 05_TDP_Measures.dax
# TARGET: _Measures table
##############################################################################
// =====================================================================
// TDP / DISTRIBUTION MEASURES  (TDP = Total Distribution Points, CPG sense)
// TDP = SUM of ACV % across all SKUs in the current grain.
// Because it is a SUM, it rolls up correctly at Brand / Category / Sub-cat /
// Chain / Zone / State / Article / Pack-size level automatically.
// =====================================================================

// Core
TDP = SUM ( 'Fact TDP'[ACV %] )                 // e.g. 3 SKUs @50% ACV => 150
ACV % = AVERAGE ( 'Fact TDP'[ACV %] )            // average shelf reach of a SKU
AIC = AVERAGE ( 'Fact TDP'[AIC] )                // Average Items Carried
Numeric Distribution = AVERAGE ( 'Fact TDP'[Numeric Distribution] )
Weighted Distribution = AVERAGE ( 'Fact TDP'[Weighted Distribution] )

// Velocity
Sales per TDP   = DIVIDE ( [NSV], [TDP] )                 // offtake NSV per distribution point
Offtake per TDP = DIVIDE ( [Total Offtake NSV], [TDP] )
Primary per TDP = DIVIDE ( [Total Primary NSV], [TDP] )

// Growth
Prev Month TDP = CALCULATE ( [TDP], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
Last Year TDP  = CALCULATE ( [TDP], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
TDP MoM Growth % = DIVIDE ( [TDP] - [Prev Month TDP], [Prev Month TDP] )
TDP YoY Growth % = DIVIDE ( [TDP] - [Last Year TDP], [Last Year TDP] )

TDP Contribution % =
DIVIDE ( [TDP],
    CALCULATE ( [TDP], ALLSELECTED ( 'Brand Master'[Brand] ), ALLSELECTED ( 'Article Master'[Article Code] ) ) )

// ---- Distribution-led vs Velocity-led diagnosis ----
// Compares Sales growth vs TDP growth to answer the leadership question:
// "is growth coming from better distribution or better velocity?"
Sales Growth vs TDP Growth = [MoM Growth %] - [TDP MoM Growth %]

Growth Driver =
VAR sg = [MoM Growth %]
VAR tg = [TDP MoM Growth %]
RETURN
    SWITCH ( TRUE (),
        ISBLANK ( sg ), "",
        sg > 0 && tg > 0 && tg >= sg, "Distribution-led",
        sg > 0 && ( ISBLANK ( tg ) || sg > tg ), "Velocity-led",
        sg > 0, "Mixed",
        "Declining"
    )

// ---- Opportunity quadrants (use as a flag column in the article table) ----
// Benchmarks = median across visible articles.
_Median Sales per TDP =
MEDIANX ( ALLSELECTED ( 'Article Master'[Article Code] ), [Sales per TDP] )
_Median TDP =
MEDIANX ( ALLSELECTED ( 'Article Master'[Article Code] ), [TDP] )

TDP Opportunity Quadrant =
VAR spt = [Sales per TDP]
VAR t = [TDP]
VAR medSPT = [_Median Sales per TDP]
VAR medTDP = [_Median TDP]
RETURN
    SWITCH ( TRUE (),
        t < medTDP && spt >= medSPT, "Low TDP / High Velocity — EXPAND",
        t >= medTDP && spt < medSPT, "High TDP / Low Velocity — FIX PRODUCTIVITY",
        t >= medTDP && spt >= medSPT, "Strong — PROTECT",
        "Low / Low — REVIEW"
    )

// Market share vs TDP link (uses Nielsen share at the matching grain)
MS vs TDP Index = DIVIDE ( [Market Share %] * 100, [TDP] )

# STEP 07/14  --  SOURCE FILE: 06_DataQuality_Measures.dax
# TARGET: _Measures table
##############################################################################
// =====================================================================
// DATA QUALITY MEASURES  (feed the Data Quality Check page)
// Each returns an impacted-row count. Build a table visual with a manual
// "Check" column (Enter Data) OR use the disconnected 'DQ Checks' helper
// table pattern at the bottom. Action text lives in that helper table.
// =====================================================================

// ---- Missing key fields (blank / null) on Offtake fact ----
Missing Month    = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( [MonthStart] ) ) )
Missing Chain    = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[Chain] ) || 'Fact Offtake Sales'[Chain] = "" ) )
Missing Zone     = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[Zone] ) || 'Fact Offtake Sales'[Zone] = "" ) )
Missing State    = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[State] ) || 'Fact Offtake Sales'[State] = "" ) )
Missing Brand    = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[Brand] ) || 'Fact Offtake Sales'[Brand] = "" ) )
Missing Category = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[Category] ) || 'Fact Offtake Sales'[Category] = "" ) )
Missing Article Code = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[Article Code] ) || 'Fact Offtake Sales'[Article Code] = "" ) )
Missing EAN      = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[EAN Code] ) || 'Fact Offtake Sales'[EAN Code] = "" ) )

// ---- Bad values ----
Blank NSV Values    = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', ISBLANK ( 'Fact Offtake Sales'[Offtake NSV] ) ) )
Negative Sales Values = CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ), FILTER ( 'Fact Offtake Sales', 'Fact Offtake Sales'[Offtake NSV] < 0 ) )

// ---- Duplicates ----
Duplicate Store Count =
SUMX ( FILTER ( SUMMARIZE ( 'Store Master', 'Store Master'[Store Code], "n", COUNTROWS ( 'Store Master' ) ), [n] > 1 ), [n] - 1 )
Duplicate Article Count =
SUMX ( FILTER ( SUMMARIZE ( 'Article Master', 'Article Master'[Article Code], "n", COUNTROWS ( 'Article Master' ) ), [n] > 1 ), [n] - 1 )

// ---- Unmapped keys (in fact but missing from master) ----
Unmapped Chain =
CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ),
    FILTER ( 'Fact Offtake Sales', NOT ( 'Fact Offtake Sales'[Chain] IN VALUES ( 'Chain Master'[Chain] ) ) ) )
Unmapped Brand =
CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ),

CALCULATE ( COUNTROWS ( 'Fact Offtake Sales' ),
    FILTER ( 'Fact Offtake Sales', NOT ( 'Fact Offtake Sales'[Zone] IN VALUES ( 'Zone State Master'[Zone] ) ) ) )

// ---- Coverage gaps in supporting facts ----
// D2: true set difference (EXCEPT), not a count subtraction. Counting
// distinctA - distinctB is wrong whenever B contains items absent from A
// (it under-counts and can even go negative). EXCEPT returns exactly the
// keys present in the first table but missing from the second.
Missing TDP =

    VALUES ( 'Fact Offtake Sales'[Article Code] ),
    VALUES ( 'Fact TDP'[Article Code] ) ) )

    VALUES ( 'Category Master'[Nielsen Category] ),
    VALUES ( 'Fact Nielsen'[Nielsen Category] ) ) )
Missing PnL Assumptions =
COUNTROWS ( EXCEPT (
    VALUES ( 'Fact Offtake Sales'[Month] ),
    VALUES ( 'Assumption Table'[Month] ) ) )

// ---- Overall health score (share of clean rows) ----
Total Fact Rows = COUNTROWS ( 'Fact Offtake Sales' )

// D1: count DISTINCT impacted rows. Adding the per-field miss counts double-
// counts a row that misses several fields, so the old score could exceed 100%
// of rows or go negative. Negative NSV is a VALID return / credit note per the
// business rule and is deliberately NOT treated as an issue here.
Rows With Any Issue =
CALCULATE (
    COUNTROWS ( 'Fact Offtake Sales' ),
    FILTER ( 'Fact Offtake Sales',
        ISBLANK ( 'Fact Offtake Sales'[Chain] )        || 'Fact Offtake Sales'[Chain] = ""
     || ISBLANK ( 'Fact Offtake Sales'[Zone] )         || 'Fact Offtake Sales'[Zone] = ""
     || ISBLANK ( 'Fact Offtake Sales'[State] )        || 'Fact Offtake Sales'[State] = ""
     || ISBLANK ( 'Fact Offtake Sales'[Brand] )        || 'Fact Offtake Sales'[Brand] = ""
     || ISBLANK ( 'Fact Offtake Sales'[Category] )     || 'Fact Offtake Sales'[Category] = ""
     || ISBLANK ( 'Fact Offtake Sales'[Article Code] ) || 'Fact Offtake Sales'[Article Code] = ""
     || ISBLANK ( 'Fact Offtake Sales'[EAN Code] )     || 'Fact Offtake Sales'[EAN Code] = ""
     || ISBLANK ( 'Fact Offtake Sales'[Offtake NSV] ) ) )

// Retained for the detail visual, but NO LONGER the health denominator (it
// double-counts multi-issue rows). Negative NSV removed per the business rule.
Total DQ Issues =
[Missing Chain] + [Missing Zone] + [Missing State] + [Missing Brand]
    + [Missing Category] + [Missing Article Code] + [Missing EAN]
    + [Blank NSV Values]
    + [Unmapped Chain] + [Unmapped Brand] + [Unmapped Category] + [Unmapped Zone]

Data Health % = 1 - DIVIDE ( [Rows With Any Issue], [Total Fact Rows] )

// =====================================================================
// SHIP-TO PRIMARY ALLOCATION MEASURES
// Primary is driven onto Chains by the secondary/offtake contribution %,
// month-on-month, via 'Primary Allocation Map' (+ optional manual
// 'Primary Allocation Override'). Nothing is hardcoded in the measures.
//
// Two modes are supported:
//  (A) Feed already pre-split to chain (today's data): the allocated NSV is
//      in 'Fact Primary ShipTo'[Primary NSV] -> use [Ship-to Primary NSV].
//  (B) Feed arrives un-split at Ship-To x Brand total: multiply that total by
//      the chain Cont% -> use [Allocated Primary (via Cont%)].
// =====================================================================

// ---------- (A) Already-allocated primary ----------
Ship-to Primary NSV = SUM ( 'Fact Primary ShipTo'[Primary NSV] )
Ship-to Primary MRP = SUM ( 'Fact Primary ShipTo'[MRP Value] )

// ---------- Allocation contribution % (dynamic, month-wise) ----------
// Override wins; else the secondary-derived map value; weighted by primary so
// it aggregates sensibly above a single Ship-To x Chain x Brand cell.
Primary Allocation Cont% =
VAR _override =
    CALCULATE (
        MAX ( 'Primary Allocation Override'[Override Cont%] ),
        FILTER (
            'Primary Allocation Override',
            ( 'Primary Allocation Override'[Month] = SELECTEDVALUE ( 'Date Table'[Month] )
                || 'Primary Allocation Override'[Month] = "ALL" )
                && ( 'Primary Allocation Override'[Ship To Name] = SELECTEDVALUE ( 'Ship-To Master'[Ship To Name] )
                    || 'Primary Allocation Override'[Ship To Name] = "ALL" )
                && ( 'Primary Allocation Override'[Chain] = SELECTEDVALUE ( 'Chain Master'[Chain] )
                    || 'Primary Allocation Override'[Chain] = "ALL" )
                && ( 'Primary Allocation Override'[Brand] = SELECTEDVALUE ( 'Brand Master'[Brand] )
                    || 'Primary Allocation Override'[Brand] = "ALL" )
        )
    )
VAR _mapWeighted =
    DIVIDE (
        SUMX ( 'Primary Allocation Map', 'Primary Allocation Map'[Cont%] ),
        COUNTROWS ( 'Primary Allocation Map' )
    )
RETURN
    IF ( NOT ISBLANK ( _override ), DIVIDE ( _override, 100 ), _mapWeighted )

// ---------- (B) Compute allocation from an un-split raw primary ----------
// Point [Raw ShipTo Primary] at your un-split feed if/when one exists.
// By default it falls back to the already-allocated NSV so the measure is safe.
Raw ShipTo Primary =
COALESCE (
    // <-- replace with SUM('Raw Primary'[Primary NSV]) when an un-split feed is added
    BLANK (),
    [Ship-to Primary NSV]
)

Allocated Primary (via Cont%) =
SUMX (
    SUMMARIZE (
        'Primary Allocation Map',
        'Primary Allocation Map'[Ship To Name],
        'Primary Allocation Map'[Chain],
        'Primary Allocation Map'[Brand]
    ),
    VAR _cont = [Primary Allocation Cont%]
    VAR _raw =
        CALCULATE (
            [Raw ShipTo Primary],
            ALL ( 'Chain Master'[Chain] )   // raw is at ship-to x brand (chain-agnostic)
        )
    RETURN _raw * _cont
)

// ---------- Secondary / Offtake at the same grain ----------
Secondary Offtake NSV = [Total Offtake NSV]
Secondary Offtake Qty = [Total Offtake Qty]

// ---------- Article inclusion: offtake this month OR previous month ----------
// Primary can lead/lag secondary by a month, so an article counts as "active"
// if it had offtake in the selected month or the one before.
Article Has Offtake (Curr or Prev) =
VAR _curr = [Total Offtake NSV]
VAR _prev = CALCULATE ( [Total Offtake NSV], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
RETURN IF ( ( COALESCE ( _curr, 0 ) <> 0 ) || ( COALESCE ( _prev, 0 ) <> 0 ), 1, 0 )

// Primary restricted to active articles (current or previous-month offtake).
Ship-to Primary NSV (Active Articles) =
CALCULATE (
    [Ship-to Primary NSV],
    FILTER (
        VALUES ( 'Article Master'[Article Code] ),
        [Article Has Offtake (Curr or Prev)] = 1
    )
)

// ---------- Primary vs Offtake on the allocated basis ----------
Allocated Primary vs Offtake Gap = [Ship-to Primary NSV] - [Secondary Offtake NSV]
Allocated Primary vs Offtake Gap % =
DIVIDE ( [Ship-to Primary NSV] - [Secondary Offtake NSV], [Secondary Offtake NSV] )
Primary to Offtake Ratio = DIVIDE ( [Ship-to Primary NSV], [Secondary Offtake NSV] )

// ---------- MoM movement on allocated primary ----------
Ship-to Primary Prev Month = CALCULATE ( [Ship-to Primary NSV], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
Ship-to Primary MoM % =
DIVIDE ( [Ship-to Primary NSV] - [Ship-to Primary Prev Month], [Ship-to Primary Prev Month] )
Ship-to Primary YoY % =
DIVIDE (
    [Ship-to Primary NSV] - CALCULATE ( [Ship-to Primary NSV], DATEADD ( 'Date Table'[Date], -12, MONTH ) ),
    CALCULATE ( [Ship-to Primary NSV], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
)

// ---------- Display helpers ----------
Ship-to Primary NSV Label =
VAR v = [Ship-to Primary NSV]
RETURN IF ( ABS ( v ) >= 10000000,
    FORMAT ( DIVIDE ( v, 10000000 ), "0.00" ) & " Cr",
    FORMAT ( DIVIDE ( v, 100000 ), "0.0" ) & " L" )

Allocation Health Check =
// distributor rows whose Cont% across chains does not sum to ~100% in a month
VAR _bad =
    COUNTROWS (
        FILTER (
            SUMMARIZE (
                'Primary Allocation Map',
                'Primary Allocation Map'[Month],
                'Primary Allocation Map'[Ship To Name],
                'Primary Allocation Map'[Brand],
                "tot", SUM ( 'Primary Allocation Map'[Cont%] )
            ),
            [tot] < 0.99 || [tot] > 1.01
        )
    )
RETURN COALESCE ( _bad, 0 )

// =====================================================================
// ARTICLE-LEVEL ALLOCATION — OFFTAKE ELIGIBILITY GATE
// =====================================================================
// Rule: do NOT allocate distributor primary to a Chain x Brand x Article x
// Month unless that exact Chain x Brand x Article shows SECONDARY OFFTAKE in
// the same month (M) or the next month (M+1, distributor->chain TAT).
// Otherwise BLOCK it as an exception. Never force allocation to an unsupported
// chain/article just to make the distributor total reach 100%.
//
// Reconciliation under the gate (per Month x Distributor x Brand):
//     Original Primary = Allocated (eligible) + Blocked (no offtake evidence)
// The ALLOCATED portion ties to the eligible-supported primary exactly; the
// BLOCKED portion is reported separately (not hidden, not forced).
//
// Tables used:
//   'Fact Offtake Sales'   = secondary/offtake (validation base)  [present]
//   'Fact Primary Article' = File 2 distributor article-wise primary [PENDING]
// Article key priority: EAN -> Article Code -> cleaned Article Description.
// Grain of these measures: Month x Distributor x Chain x Brand x Article.
// =====================================================================

// ---------- Secondary evidence at the Chain x Brand x Article grain ----------
Offtake NSV (CBA, M) = [Total Offtake NSV]   // current month, current C/B/A context

Offtake NSV (CBA, M+1) =
CALCULATE ( [Total Offtake NSV], DATEADD ( 'Date Table'[Date], 1, MONTH ) )

// brand-level evidence at the chain (to tell "brand not listed" from "article not listed")
Offtake NSV (Chain-Brand, M or M+1) =
CALCULATE (
    [Total Offtake NSV] + [Offtake NSV (CBA, M+1)],
    ALL ( 'Article Master'[Article Code] ), ALL ( 'Article Master'[EAN Code] )
)

// first month this Chain x Brand x Article ever shows offtake (for new-article logic)
First Offtake Month (CBA) =
CALCULATE (
    MIN ( 'Fact Offtake Sales'[MonthStart] ),
    ALLEXCEPT ( 'Fact Offtake Sales',
        'Fact Offtake Sales'[Chain], 'Fact Offtake Sales'[Brand], 'Fact Offtake Sales'[Article Code] )
)

// ---------- Eligibility ----------
Article Eligible =
VAR _m   = COALESCE ( [Offtake NSV (CBA, M)], 0 )
VAR _m1  = COALESCE ( [Offtake NSV (CBA, M+1)], 0 )
RETURN IF ( _m <> 0 || _m1 <> 0, 1, 0 )

Eligibility Status =
VAR _m   = COALESCE ( [Offtake NSV (CBA, M)], 0 )
VAR _m1  = COALESCE ( [Offtake NSV (CBA, M+1)], 0 )
VAR _brand = COALESCE ( [Offtake NSV (Chain-Brand, M or M+1)], 0 )
RETURN
    SWITCH ( TRUE (),
        _m  <> 0, "Eligible",                    // offtake in same month
        _m1 <> 0, "Eligible due to TAT",         // offtake only next month
        _brand = 0, "Brand not listed",          // chain has no offtake for this brand
        "Article not listed"                     // brand listed at chain, this article not
    )

// New-article one-month-prior flag: offtake first appears in M+1 (so primary
// allowed in M). True when current month has no offtake but M+1 does and M+1 is
// the first offtake month.
New Article Flag =
VAR _m  = COALESCE ( [Offtake NSV (CBA, M)], 0 )
VAR _m1 = COALESCE ( [Offtake NSV (CBA, M+1)], 0 )
VAR _first = [First Offtake Month (CBA)]
VAR _nextMonth = EOMONTH ( MAX ( 'Date Table'[MonthStart] ), 1 ) + 1
RETURN IF ( _m = 0 && _m1 <> 0 && _first = _nextMonth, "New Article - primary held 1 month prior", "" )

Primary Month Maintained =
IF ( [New Article Flag] <> "", FORMAT ( MAX ( 'Date Table'[MonthStart] ), "mmm'yy" ), "" )

// ---------- Chain-allocated primary (Step 2 result, from Cont%) ----------
// Distributor-brand primary split to chain by the secondary Cont% (built earlier).
Chain Allocated Primary NSV = [Ship-to Primary NSV]   // already chain-split in File 1

// ---------- Article ratio among ELIGIBLE articles only ----------
// Denominator = secondary of eligible articles in this Chain x Dist x Brand x Month,
// so eligible articles absorb the chain primary; blocked ones get nothing.
_Eligible Article Secondary (Chain-Brand) =
CALCULATE (
    [Total Offtake NSV] + [Offtake NSV (CBA, M+1)],
    FILTER ( VALUES ( 'Article Master'[Article Code] ), [Article Eligible] = 1 ),
    VALUES ( 'Chain Master'[Chain] ), VALUES ( 'Brand Master'[Brand] )
)
Article Allocation Ratio % =
IF ( [Article Eligible] = 1,
    DIVIDE ( COALESCE ( [Offtake NSV (CBA, M)], 0 ) + COALESCE ( [Offtake NSV (CBA, M+1)], 0 ),
             [_Eligible Article Secondary (Chain-Brand)] ),
    0 )

// ---------- Allocation outputs ----------
Allocated Article Primary NSV =
IF ( [Article Eligible] = 1,
    [Chain Allocated Primary NSV] * [Article Allocation Ratio %],
    0 )

// Allocated Qty / MRP follow the same ratio (need File 2 Qty/MRP)
Allocated Article Primary Qty =
IF ( [Article Eligible] = 1,
    CALCULATE ( SUM ( 'Fact Primary Article'[Primary Qty] ) ) * [Article Allocation Ratio %],
    0 )

// Primary that is NOT allocated because there is no offtake evidence at all
// for the brand/article in this chain (the legitimate blocked bucket).
Blocked Primary NSV =
VAR _hasEligible = [_Eligible Article Secondary (Chain-Brand)]
RETURN IF ( COALESCE ( _hasEligible, 0 ) = 0, [Chain Allocated Primary NSV], 0 )

Allocation Status =
SWITCH ( TRUE (),
    [Article Eligible] = 1, "Allocated",
    [Eligibility Status] = "Brand not listed",   "Blocked - Brand Not Listed",
    [Eligibility Status] = "Article not listed", "Blocked - Article Not Listed",
    "Blocked - No Offtake Evidence" )

Exception Reason =
SWITCH ( TRUE (),
    [Article Eligible] = 1, "",
    [Eligibility Status] = "Brand not listed",   "Brand has no offtake in this chain (M or M+1)",
    [Eligibility Status] = "Article not listed", "Article has no offtake in this chain (M or M+1)",
    "No secondary offtake found in M or M+1" )

// =====================================================================
// QC — reconciliation separates VALID allocation from BLOCKED/UNALLOCATED
// =====================================================================
QC Orig Primary (Dist-Brand) =
CALCULATE ( [Ship-to Primary NSV],
    ALLSELECTED ( 'Chain Master'[Chain] ), ALLSELECTED ( 'Article Master'[Article Code] ) )
QC Allocated Primary Total = [Allocated Article Primary NSV]
QC Blocked Primary Total   = [Blocked Primary NSV]
QC Reconciliation Variance =
[QC Orig Primary (Dist-Brand)] - ( [QC Allocated Primary Total] + [QC Blocked Primary Total] )
// ^ MUST be 0:  Original = Allocated + Blocked  (no value lost, none forced)

QC Blocked Article Count =
CALCULATE ( DISTINCTCOUNT ( 'Article Master'[Article Code] ),
    FILTER ( VALUES ( 'Article Master'[Article Code] ), [Article Eligible] = 0 ) )
QC Allocated Article Count =
CALCULATE ( DISTINCTCOUNT ( 'Article Master'[Article Code] ),
    FILTER ( VALUES ( 'Article Master'[Article Code] ), [Article Eligible] = 1 ) )
QC Mapping Coverage % =
DIVIDE ( [QC Allocated Primary Total], [QC Orig Primary (Dist-Brand)] )
QC Blocked Coverage % =
DIVIDE ( [QC Blocked Primary Total], [QC Orig Primary (Dist-Brand)] )

// ---------- Helpers for the New Article Tracker export ----------
Allowed Primary Month (date) = EOMONTH ( [First Offtake Month (CBA)], -1 ) + 1

First Offtake NSV =
CALCULATE ( [Total Offtake NSV],

        'Date Table'[MonthStart] = [First Offtake Month (CBA)] ) )

Primary NSV in Allowed Month =
CALCULATE ( [Chain Allocated Primary NSV],
    FILTER ( ALL ( 'Date Table' ),
        'Date Table'[MonthStart] = [Allowed Primary Month (date)] ) )

// =====================================================================
// TOT% MEASURES  (Trade Offer Terms % / On-Invoice Margin Pass-on %)
//
// TOT% = SUM(Pass-on Value) / SUM(MRP) -- never a simple average of
// row-level TOT% percentages. Pass-on Value is sourced per row with a
// 3-TIER PRIORITY (mirrors dashboard/scripts/build_dashboard_data.py's
// tot_block() exactly, so a chain/category's TOT% reads the same on the
// HTML dashboard and here):
//
//   1. SOURCE      -- 'Fact Primary Article'[Avg TOT] (query 16 — the
//                     Primary file's own Customer x Article-grain TOT%, a
//                     0-1 fraction). Pass-on Value = Primary MRP x Avg TOT.
//   2. ACTUAL TAX  -- if Avg TOT is blank/invalid, use the row's actual
//                     [Primary Tax Amount] ("Inv. Tax Amount(LOC)"):
//                     Pass-on Value = MRP - NSV - Tax.
//   3. GST FALLBACK -- only if BOTH are blank/invalid: estimate Tax from
//                     Category + cutover date via 'GST Rate QC Table'
//                     (query 37, editable PowerBI/SeedData/Masters/
//                     GST_Rate_QC_Table.csv) and 'GST Config' (query 38,
//                     GST_Config.csv -- default cutover 2025-09-22, the GST
//                     Council's confirmed GST 2.0 effective date, editable
//                     single cell; a per-category override lives in the QC
//                     table's Effective_From column).
//
// GST_Rate_QC_Table.csv is FALLBACK ONLY: it has zero effect on TOT% for any
// row where Avg TOT or Primary Tax Amount is present -- [TOT Rows GST
// Fallback] / [TOT Fallback Pct of MRP] below show exactly how much (if
// any) of the current data actually depends on it. Every row in that CSV
// starts Finance_Approved = "Pending" and several are Confidence = "Low"
// (no official HSN-code-level source was available) -- verify against
// Honasa's Finance/Tax records before treating any fallback-tier TOT% as
// final; [Finance Approved Count] surfaces the sign-off rollup.
// =====================================================================
