// =====================================================================
// DATE TABLE  —  create via:  Modeling > New table  (paste this DAX)
// Marked as the model's date table (Table tools > Mark as date table > [Date]).
// Covers FY Apr->Mar (Indian financial year). Spans 3 yrs back to next FY so
// YoY and forecast months always resolve. Adjust the start/end if needed.
// =====================================================================
Date Table =
VAR _Start = DATE(2024, 4, 1)        // first day of earliest FY you report
VAR _End   = DATE(2028, 3, 31)       // last day of the latest FY you forecast
RETURN
ADDCOLUMNS(
    CALENDAR(_Start, _End),
    "MonthStart",      EOMONTH([Date], -1) + 1,
    "Year",            YEAR([Date]),
    "Month No",        MONTH([Date]),
    "Month Name",      FORMAT([Date], "mmm"),
    // Indian FY: Apr=1 ... Mar=12
    "FY Month No",     IF(MONTH([Date]) >= 4, MONTH([Date]) - 3, MONTH([Date]) + 9),
    "FY Year",
        VAR y = YEAR([Date])
        RETURN IF(MONTH([Date]) >= 4,
                 FORMAT(MOD(y, 100), "00") & "-" & FORMAT(MOD(y + 1, 100), "00"),
                 FORMAT(MOD(y - 1, 100), "00") & "-" & FORMAT(MOD(y, 100), "00")),
    "Quarter",
        VAR fm = IF(MONTH([Date]) >= 4, MONTH([Date]) - 3, MONTH([Date]) + 9)
        RETURN "Q" & ROUNDUP(fm / 3, 0),
    // label used everywhere in the deck, e.g. May'26
    "Month",           FORMAT([Date], "mmm") & "'" & FORMAT([Date], "yy"),
    "Month Year Sort", YEAR([Date]) * 100 + MONTH([Date]),
    "Is Future",       IF([Date] > TODAY(), TRUE(), FALSE())
)

// -----------------------------------------------------------------
// AFTER creating: set Sort-by columns
//   [Month]      sort by [Month Year Sort]
//   [Month Name] sort by [Month No]
//   [Quarter]    sort by [FY Month No]
// Relationships (single direction, 1 -> *):
//   Date Table[MonthStart] -> Fact Offtake Sales[MonthStart]
//   Date Table[Date]       -> Fact Primary Sales[Week Start Date]
//   Date Table[MonthStart] -> Fact P&L[MonthStart]
//   Date Table[MonthStart] -> Fact Nielsen[MonthStart]
//   Date Table[MonthStart] -> Fact TDP[MonthStart]
//   Date Table[MonthStart] -> Targets[MonthStart]
// -----------------------------------------------------------------

// =====================================================================
// CORE MEASURES  —  create a "_Measures" table (Enter Data, one blank col)
// and add each measure below to it. Every division uses DIVIDE().
// Convention: "NSV" headline = Offtake NSV (matches the leadership deck).

// ---------- BASE TOTALS ----------
Total Offtake NSV = SUM ( 'Fact Offtake Sales'[Offtake NSV] )
Total Offtake Qty = SUM ( 'Fact Offtake Sales'[Offtake Qty] )
// The monthly Article fact is loaded; the legacy weekly Primary Sales query is deferred.
// Primary stays gross billing, including Reliance Brand Counter.
Total Primary NSV = SUM ( 'Fact Primary Article'[Primary NSV] )
Total Primary Qty = SUM ( 'Fact Primary Article'[Primary Qty] )
Total MRP Sales   = SUM ( 'Fact Offtake Sales'[MRP Sales] )

// ---------- RELIANCE BRAND COUNTER (RBC) ISOLATION ----------
Total MT NSV =
    CALCULATE (
        [Total Offtake NSV],
        'Chain Master'[Channel] = "MT"
    )

Total RBC NSV =
    CALCULATE (
        [Total Offtake NSV],
        'Chain Master'[Channel] = "RBC"
    )

RBC % of Total =
    DIVIDE (
        [Total RBC NSV],
        [Total MT NSV] + [Total RBC NSV]
    )

Combined MT Total NSV =
    [Total MT NSV] + [Total RBC NSV]

Is RBC =
    IF (
        HASONEVALUE ( 'Chain Master'[Channel] ) && VALUES ( 'Chain Master'[Channel] ) = "RBC",
        "RBC (Brand Counter)",
        "MT (Modern Trade)"
    )

// Headline NSV used by most visuals (Offtake basis). Swap to Primary if needed.
NSV = [Total MT NSV]

// ---------- INDIAN FORMAT HELPERS (display measures) ----------
NSV (Cr) = DIVIDE ( [NSV], 10000000 )
NSV (Lacs) = DIVIDE ( [NSV], 100000 )
// Smart label: Cr if >= 1 Cr else Lacs
NSV Label =
VAR v = [NSV]
RETURN
    IF (
        ABS ( v ) >= 10000000,
        FORMAT ( DIVIDE ( v, 10000000 ), "0.00" ) & " Cr",
        FORMAT ( DIVIDE ( v, 100000 ), "0.0" ) & " L"
    )

// ---------- TIME INTELLIGENCE (Offtake basis) ----------
Latest Month Sales =
VAR _lastDate = MAX ( 'Date Table'[Date] )
RETURN CALCULATE ( [NSV], 'Date Table'[Date] <= _lastDate )
// In a month-grain visual [NSV] already = the selected month; the explicit
// "latest month" KPI for cards:
Latest Month NSV =
VAR _lm = CALCULATE ( MAX ( 'Fact Offtake Sales'[MonthStart] ), ALL ( 'Date Table' ) )
RETURN CALCULATE ( [NSV], 'Date Table'[MonthStart] = _lm )

Previous Month NSV =
CALCULATE ( [NSV], DATEADD ( 'Date Table'[Date], -1, MONTH ) )

MoM Growth % =
VAR cur = [NSV]
VAR prev = [Previous Month NSV]
RETURN DIVIDE ( cur - prev, prev )

Last Year Same Month NSV =
CALCULATE ( [NSV], DATEADD ( 'Date Table'[Date], -12, MONTH ) )

YoY Growth % =
DIVIDE ( [NSV] - [Last Year Same Month NSV], [Last Year Same Month NSV] )

// ---------- ROLLING AVERAGES ----------
// Averages over the months that have data: a month before the source starts is
// missing, not zero, so it must not pull the average down. A month that sold 0
// still counts (0 is not BLANK). tests/test_powerbi_rolling_average.py
L3M Average Sales =
VAR _curMonth = MAX ( 'Date Table'[MonthStart] )
RETURN
    CALCULATE (
        AVERAGEX ( FILTER ( VALUES ( 'Date Table'[MonthStart] ), NOT ISBLANK ( [NSV] ) ), [NSV] ),
        DATESINPERIOD ( 'Date Table'[Date], EOMONTH ( _curMonth, -1 ), -3, MONTH )
    )

// Averages over the months that have data: a month before the source starts is
// missing, not zero, so it must not pull the average down. A month that sold 0
// still counts (0 is not BLANK). tests/test_powerbi_rolling_average.py
L6M Average Sales =
VAR _curMonth = MAX ( 'Date Table'[MonthStart] )
RETURN
    CALCULATE (
        AVERAGEX ( FILTER ( VALUES ( 'Date Table'[MonthStart] ), NOT ISBLANK ( [NSV] ) ), [NSV] ),
        DATESINPERIOD ( 'Date Table'[Date], EOMONTH ( _curMonth, -1 ), -6, MONTH )
    )

Growth vs L3M % = DIVIDE ( [NSV] - [L3M Average Sales], [L3M Average Sales] )

// ---------- CONTRIBUTION & RANK ----------
Contribution % =
DIVIDE (
    [NSV],
    CALCULATE ( [NSV], ALLSELECTED ( 'Chain Master'[Chain] ), ALLSELECTED ( 'Brand Master'[Brand] ),
        ALLSELECTED ( 'Category Master'[Category] ) )
)

Rank by Sales =
IF ( ISINSCOPE ( 'Chain Master'[Chain] ),
    RANKX ( ALLSELECTED ( 'Chain Master'[Chain] ), [NSV],, DESC ),
    RANKX ( ALLSELECTED ( 'Brand Master'[Brand] ), [NSV],, DESC )
)

// ---------- CHAIN CONCENTRATION & TOP MOVERS ----------
Top 3 Concentration % =
VAR top3Total =
    SUMX (
        TOPN ( 3, ALLSELECTED ( 'Chain Master'[Chain] ), [Total MT NSV], DESC ),
        [Total MT NSV]
    )
RETURN
    DIVIDE ( top3Total, [Total MT NSV] )

Top Chain =
    CONCATENATEX (
        TOPN ( 1, ALLSELECTED ( 'Chain Master'[Chain] ), [Total MT NSV], DESC ),
        'Chain Master'[Chain],
        ""
    )

Top 3 Chains =
    CONCATENATEX (
        TOPN ( 3, ALLSELECTED ( 'Chain Master'[Chain] ), [Total MT NSV], DESC ),
        'Chain Master'[Chain],
        ", "
    )

// ---------- PRIMARY vs OFFTAKE ----------
Primary vs Offtake Gap = [Total Primary NSV] - [Total MT NSV]
Primary vs Offtake Gap % = DIVIDE ( [Total Primary NSV] - [Total MT NSV], [Total MT NSV] )

// ---------- GROWTH ARROW (for tables / cards) ----------
MoM Arrow =
VAR g = [MoM Growth %]
RETURN IF ( ISBLANK ( g ), "", IF ( g >= 0, "▲ ", "▼ " ) ) & FORMAT ( g, "0.0%" )

YoY Arrow =
VAR g = [YoY Growth %]
RETURN IF ( ISBLANK ( g ), "", IF ( g >= 0, "▲ ", "▼ " ) ) & FORMAT ( g, "0.0%" )

// Conditional-format colour (use in "Font color > fx > Field value")
Growth Colour =
SWITCH ( TRUE (),

    [MoM Growth %] < 0, "#C0392B",   // red negative

)

// ---------- TOP / BOTTOM MOVERS (Executive Summary) ----------
Top Growing Chain =

Top Declining Chain =

Top Growing Brand =
VAR t = TOPN ( 1, ALLSELECTED ( 'Brand Master'[Brand] ), [MoM Growth %], DESC )
RETURN CONCATENATEX ( t, 'Brand Master'[Brand], ", " )

Top Growing Category =
VAR t = TOPN ( 1, ALLSELECTED ( 'Category Master'[Category] ), [MoM Growth %], DESC )
RETURN CONCATENATEX ( t, 'Category Master'[Category], ", " )

// =====================================================================
// TARGET / ACHIEVEMENT / RUN RATE
// Mirrors scripts/build_dashboard_data.py targets_block() so the Power BI
// report and the web dashboard state the same numbers from the same rules.
// [TY Target] is defined in 03_Forecast_Measures.dax (Targets[Target NSV]).
//
// BASIS: the target file is NSV and does not name its measure. The Forecast
// page compares it to OFFTAKE, so these default to offtake. Swap
// [Total MT NSV] for [Total Primary NSV] below to move the basis --
// change it in "Actual (Basis)" only; everything else follows.

Actual (Basis) = [Total MT NSV]

// ---------- PERIOD TO DATE (only months that actually have actuals) ----------
// The PTD window is DERIVED from where actuals exist, so it widens on its own
// as each new month lands -- no hardcoded month count to maintain.
Last Actual Month =
    CALCULATE (
        MAX ( 'Date Table'[MonthStart] ),
        FILTER ( ALL ( 'Date Table' ), [Actual (Basis)] > 0 )
    )

Target PTD =
    CALCULATE (
        [TY Target],
        FILTER ( ALL ( 'Date Table' ), 'Date Table'[MonthStart] <= [Last Actual Month] )
    )

Actual PTD =
    CALCULATE (
        [Actual (Basis)],
        FILTER ( ALL ( 'Date Table' ), 'Date Table'[MonthStart] <= [Last Actual Month] )
    )

// ---------- ACHIEVEMENT ----------
Achievement % = DIVIDE ( [Actual PTD], [Target PTD] )

Gap to Target = [Actual PTD] - [Target PTD]

Gap to Target % = DIVIDE ( [Gap to Target], [Target PTD] )

Achievement Status =
    VAR _a = [Achievement %]
    RETURN
        SWITCH (
            TRUE (),
            ISBLANK ( _a ), "No target",
            _a >= 1.00, "Met",
            _a >= 0.90, "Near",
            "Missed"
        )

// ---------- RUN RATE ----------
Months Elapsed =
    CALCULATE (
        DISTINCTCOUNT ( 'Date Table'[MonthStart] ),
        FILTER ( ALL ( 'Date Table' ), 'Date Table'[MonthStart] <= [Last Actual Month] ),
        FILTER ( ALL ( 'Date Table' ), NOT ISBLANK ( [TY Target] ) )
    )

Months In FY = CALCULATE ( DISTINCTCOUNT ( Targets[MonthStart] ), ALL ( 'Date Table' ) )

Months Remaining = [Months In FY] - [Months Elapsed]

Current Run Rate = DIVIDE ( [Actual PTD], [Months Elapsed] )

// What the remaining months must average to still land the full-year target.
Required Run Rate =
    DIVIDE (
        CALCULATE ( [TY Target], ALL ( 'Date Table' ) ) - [Actual PTD],
        [Months Remaining]
    )

Run Rate Status =
    VAR _req = [Required Run Rate]
    VAR _cur = [Current Run Rate]
    VAR _ratio = DIVIDE ( _req, _cur )
    RETURN
        SWITCH (
            TRUE (),
            _cur = 0 || ISBLANK ( _ratio ), "No actuals",
            _ratio <= 1.00, "Ahead",
            _ratio <= 1.05, "On Track",
            _ratio <= 1.15, "At Risk",
            "Critical"
        )

// ---------- LIKE-FOR-LIKE YoY ----------
// A part-year FY against a full prior FY is not a YoY. Compare only the
// months the current FY actually carries.
Actual LY Same Period =
    CALCULATE (
        [Actual (Basis)],
        FILTER (
            ALL ( 'Date Table' ),
            'Date Table'[MonthStart]
                <= EDATE ( [Last Actual Month], -12 )
                && 'Date Table'[FY Year]
                    = CALCULATE ( MAX ( 'Date Table'[FY Year] ),
                        FILTER ( ALL ( 'Date Table' ),
                            'Date Table'[MonthStart] = EDATE ( [Last Actual Month], -12 ) ) )
        )
    )

YoY % (Like-for-Like) =
    DIVIDE ( [Actual PTD] - [Actual LY Same Period], [Actual LY Same Period] )

// =====================================================================
// P&L MEASURES  (actuals-first, else Assumption Table — never hardcoded)
// Logic: if a raw P&L actual exists use it; otherwise resolve the most
// specific matching row in 'Assumption Table' (Month>Chain>Brand>Category),
// falling back to "ALL". All divisions use DIVIDE().
//
// F21 (docs/PHASE_2B_FINANCIAL_CONSUMER_INVENTORY.md follow-up sweep,
// 2026-09-25): this used to COALESCE a missing month to a hardcoded 50%
// Gross Margin / 0% Trade Spend -- 'AssumptionTable.csv' only has rows for
// Apr'26 and May'26 (each remarked "Default portfolio assumption - update
// with actuals when available"), so every month since silently computed
// CM%/Gross Margin off a made-up default with no on-screen disclosure.
// Fail-closed now: a month with no Assumption Table row returns BLANK()
// all the way through Gross Margin/Trade Spend/Total Spend/Contribution
// Margin/CM% -- never a fabricated number -- and `Assumption Status` names
// the exact gap for a QC card. `scripts/check_assumption_coverage.py` is
// the release-gate half of this fix: it fails the build with
// BLOCKED_FINANCE_INPUT before a missing month can even reach here.
// =====================================================================

// ---- Helper: is there ANY Assumption Table row for the selected month? ----
// Single source of truth for "does Finance's input exist for this month",
// independent of which specific % or spend column a caller reads next --
// AssumptionTable.csv always populates GM%/Trade Spend%/Visibility/Scheme
// together per row, so one row-existence check covers all four safely.
_Assumption Available =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
    NOT ISBLANK ( _m )
        && CALCULATE ( COUNTROWS ( 'Assumption Table' ),
               FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) ) > 0

// Finance-facing QC status -- put this on a card in the P&L page so a
// missing month is visibly flagged, not silently absorbed into "0.0%".
Assumption Status =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
    SWITCH (
        TRUE (),
        ISBLANK ( _m ), "NO MONTH SELECTED",
        [_Assumption Available], "OK",
        "ASSUMPTION_MISSING — FINANCE_INPUT_REQUIRED: " & _m
            & " has no approved Assumption Table row. Gross Margin, Trade Spend, "
            & "Contribution Margin and CM% for this month are BLANK, not estimated."
    )

// ---- Helper: resolve an assumption value with ALL-fallback ----
// Returns the % / spend for the current Month x Chain x Brand x Category
// context, preferring the most specific row, else the ALL row. Returns
// BLANK() (not a hardcoded default) when the month itself has no row at
// all -- MAX()/SUM() over an empty filtered table already returns BLANK(),
// so removing the old COALESCE(...,default) is the entire fix here.
_Assumption Gross Margin % =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
VAR _ch = SELECTEDVALUE ( 'Chain Master'[Chain], "ALL" )
VAR _br = SELECTEDVALUE ( 'Brand Master'[Brand], "ALL" )
VAR _cat = SELECTEDVALUE ( 'Category Master'[Category], "ALL" )
RETURN
    CALCULATE ( MAX ( 'Assumption Table'[Gross Margin %] ),
        FILTER ( ALL ( 'Assumption Table' ),
            'Assumption Table'[Month] = _m
            && ( 'Assumption Table'[Chain] = _ch || 'Assumption Table'[Chain] = "ALL" )
            && ( 'Assumption Table'[Brand] = _br || 'Assumption Table'[Brand] = "ALL" )
            && ( 'Assumption Table'[Category] = _cat || 'Assumption Table'[Category] = "ALL" ) ) )

_Assumption Trade Spend % =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
    CALCULATE ( MAX ( 'Assumption Table'[Trade Spend %] ),
        FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) )

_Assumption Visibility Spend =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
    CALCULATE ( SUM ( 'Assumption Table'[Visibility Spend] ),
        FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) )

_Assumption Scheme Spend =
VAR _m = SELECTEDVALUE ( 'Date Table'[Month] )
RETURN
    CALCULATE ( SUM ( 'Assumption Table'[Scheme Spend] ),
        FILTER ( ALL ( 'Assumption Table' ), 'Assumption Table'[Month] = _m ) )

// ---- P&L line items ----
Net Sales = [NSV]                                   // NSV = Net Sales Value (offtake)
Gross Sales = COALESCE ( [Total MRP Sales], DIVIDE ( [NSV], 1 - 0.0 ) )  // use MRP Sales as gross proxy

// Explicit IF guards below (not just relying on DAX's number*BLANK=BLANK /
// number-BLANK=number arithmetic quirks) so the fail-closed behavior is
// stated plainly for a Finance/Power BI reviewer, not implied. The
// subtraction/addition below would otherwise silently treat a genuinely
// missing month as a real zero (BLANK acts as 0 under + and -, unlike *).
Gross Margin = IF ( [_Assumption Available], [NSV] * [_Assumption Gross Margin %], BLANK () )
Gross Margin % = DIVIDE ( [Gross Margin], [NSV] )

Trade Spend = IF ( [_Assumption Available], [NSV] * [_Assumption Trade Spend %], BLANK () )
Visibility Spend = [_Assumption Visibility Spend]
Scheme Spend = [_Assumption Scheme Spend]
Total Spend = IF ( [_Assumption Available], [Trade Spend] + [Visibility Spend] + [Scheme Spend], BLANK () )

Spend % of NSV = DIVIDE ( [Total Spend], [NSV] )

Contribution Margin = IF ( [_Assumption Available], [Gross Margin] - [Total Spend], BLANK () )
CM % = DIVIDE ( [Contribution Margin], [NSV] )

// ---- ranking helper for the profitability page ----
Chain CM Rank =
RANKX ( ALLSELECTED ( 'Chain Master'[Chain] ), [Contribution Margin],, DESC )

// ---- "growth profitable or only sales-led?" flag ----
// F21: cm blank (missing assumption month) must not fall through to the
// generic "Needs attention" catch-all -- blank compares as 0 numerically,
// so g>0 && cm>=0.25 would silently read as false and misclassify a
// missing month as a real, bad one instead of a data gap.
Profitability Flag =
VAR g = [MoM Growth %]
VAR cm = [CM %]
RETURN
    SWITCH ( TRUE (),
        ISBLANK ( cm ), "ASSUMPTION_MISSING",
        g > 0 && cm >= 0.25, "Profitable Growth",
        g > 0 && cm < 0.25,  "Sales-led (watch margin)",
        g <= 0 && cm >= 0.25, "Margin-protected decline",
        "Needs attention"
    )

// =====================================================================
// FORECAST MEASURES  —  TY-TARGET DRIVEN  (replaces FY26 uplift logic)
// =====================================================================
// FY27 Forecast IS the TY target shared by the sales team (table 'Targets').
// The monthly target is allocated down to Chain / Zone / Brand / Category /
// Sales Person using each dimension's FY26 actual-offtake share, so EVERY
// breakdown still sums EXACTLY to the TY target total. Records with no
// mapping are NOT dropped — they bucket into "Unmapped".
//
// Allocation identity:  Σ(all dims) Forecast = TY Target  (variance = 0).
// =====================================================================

// ---------- Authoritative totals ----------
TY Target = CALCULATE ( SUM ( Targets[Target NSV] ), Targets[FY Year] = "26-27" )
TY Target (Cr) = DIVIDE ( [TY Target], 10000000 )

FY26 Actual Offtake = CALCULATE ( [Total Offtake NSV], 'Date Table'[FY Year] = "25-26" )
FY26 Actual Offtake (Cr) = DIVIDE ( [FY26 Actual Offtake], 10000000 )

// ---------- Allocation key = dimension share of FY26 offtake ----------
// Denominator is the FY26 offtake of the *whole* visible selection, so the
// shares across the axis sum to 1 and the target is fully distributed.
_FY26 Total Offtake (All Dims) =
CALCULATE (
    [FY26 Actual Offtake],
    ALLSELECTED ( 'Chain Master'[Chain] ),
    ALLSELECTED ( 'Brand Master'[Brand] ),
    ALLSELECTED ( 'Category Master'[Category] ),
    ALLSELECTED ( 'Zone State Master'[Zone] ),
    ALLSELECTED ( 'Sales Team Mapping'[Sales Person] )
)

FY26 Allocation Share = DIVIDE ( [FY26 Actual Offtake], [_FY26 Total Offtake (All Dims)] )

// ---------- FY27 Forecast (= allocated TY target) ----------
// At grand total the share = 1 so Forecast = TY Target exactly.
FY27 Forecast =
SUMX (
    VALUES ( 'Date Table'[MonthStart] ),
    VAR _monthTarget = [TY Target]
    VAR _share = [FY26 Allocation Share]
    RETURN _monthTarget * _share
)
FY27 Forecast (Cr) = DIVIDE ( [FY27 Forecast], 10000000 )

// Final Forecast keeps the old name used elsewhere, now target-driven.
Final Forecast = [FY27 Forecast]

// ---------- Unmapped bucket (do NOT exclude) ----------
// FY26 offtake whose Chain / Brand / Category is absent from the masters,
// expressed as a share, so the target portion it represents shows as Unmapped.
_FY26 Offtake Unmapped =
CALCULATE (
    [FY26 Actual Offtake],
    FILTER ( 'Fact Offtake Sales',
        NOT ( 'Fact Offtake Sales'[Chain]    IN VALUES ( 'Chain Master'[Chain] ) )
        || NOT ( 'Fact Offtake Sales'[Brand]    IN VALUES ( 'Brand Master'[Brand] ) )
        || NOT ( 'Fact Offtake Sales'[Category] IN VALUES ( 'Category Master'[Category] ) )
    )
)
Forecast - Unmapped =
[TY Target] * DIVIDE ( [_FY26 Offtake Unmapped], [_FY26 Total Offtake (All Dims)] )

// ---------- Sales-person ownership (via Sales Team Mapping x store offtake) ----------
// SO FY26 offtake = Σ over their stores of store offtake × Cont%.
SO FY26 Offtake =
SUMX (
    'Sales Team Mapping',
    VAR _store = 'Sales Team Mapping'[Store Code]
    VAR _w = 'Sales Team Mapping'[Cont%]
    RETURN _w * CALCULATE ( [FY26 Actual Offtake],
        'Fact Offtake Sales'[Store Code] = _store )
)
SO Forecast (Owned Target) =
[TY Target] * DIVIDE ( [SO FY26 Offtake],
    CALCULATE ( [SO FY26 Offtake], ALLSELECTED ( 'Sales Team Mapping'[Sales Person] ) ) )

// ---------- Gap vs FY26 & required growth ----------
Gap vs FY26 = [FY27 Forecast] - [FY26 Actual Offtake]
Required Growth % = DIVIDE ( [FY27 Forecast] - [FY26 Actual Offtake], [FY26 Actual Offtake] )

// ---------- Accuracy (once FY27 actuals start landing) ----------
Has Actual = IF ( [Total Offtake NSV] > 0, TRUE (), FALSE () )
Actual NSV = [Total Offtake NSV]
Forecast vs Actual = [Actual NSV] - [Final Forecast]
Forecast vs Actual % = DIVIDE ( [Actual NSV] - [Final Forecast], [Final Forecast] )
Forecast Accuracy % =
IF ( [Has Actual],
    1 - DIVIDE ( ABS ( [Actual NSV] - [Final Forecast] ), [Actual NSV] ),
    BLANK () )

// =====================================================================
// SHIP-TO PRIMARY ALLOCATION MEASURES
// Primary is driven onto Chains by the secondary/offtake contribution %,
// month-on-month, via 'Primary Allocation Map' (+ optional manual
// 'Primary Allocation Override'). Nothing is hardcoded in the measures.
//
// Two modes are supported:
//  (A) Feed already pre-split to chain (today's data): the allocated NSV is
//      in 'Fact Primary ShipTo'[Primary NSV] -> use [Ship-to Primary NSV].
//  (B) Feed arrives un-split at Ship-To x Brand total: multiply that total by
//      the chain Cont% -> use [Allocated Primary (via Cont%)].
//
// !! DOUBLE-ALLOCATION LANDMINE (Phase 2B-B F10, docs/PHASE_2B_FINANCIAL_
// CONSUMER_INVENTORY.md) -- read before adding [Allocated Primary (via
// Cont%)] to ANY visual, card or table:
//   Today's data is mode (A) -- already chain-split. [Raw ShipTo Primary]
//   is a placeholder that COALESCEs to [Ship-to Primary NSV] until a real
//   un-split feed exists. That means [Allocated Primary (via Cont%)]
//   currently multiplies an ALREADY chain-split NSV by the Cont% allocation
//   a SECOND time -- a silent double-application producing a wrong nonzero
//   per chain, not a gap that would be obvious on the page. It is not used
//   anywhere today: PowerBI/docs/PageLayouts.md's Page 2B (the only page
//   built from this file's measures) uses [Ship-to Primary NSV] exclusively
//   for every KPI, visual and table column -- confirmed by direct search,
//   2026-09-24. Do NOT wire it into a report page while mode (A) is still
//   the live data shape. It only becomes safe to surface once the "<--
//   replace with SUM('Raw Primary'[Primary NSV])" line below is actually
//   replaced with a real un-split feed (mode B).
// =====================================================================

// ---------- (A) Already-allocated primary ----------
Ship-to Primary NSV = SUM ( 'Fact Primary ShipTo'[Primary NSV] )
Ship-to Primary MRP = SUM ( 'Fact Primary ShipTo'[MRP Value] )

// ---------- Allocation contribution % (dynamic, month-wise) ----------
// Override wins; else the secondary-derived map value; weighted by primary so
// it aggregates sensibly above a single Ship-To x Chain x Brand cell.
Primary Allocation Cont% =
VAR _override =
    CALCULATE (
        MAX ( 'Primary Allocation Override'[Override Cont%] ),
        FILTER (
            'Primary Allocation Override',
            ( 'Primary Allocation Override'[Month] = SELECTEDVALUE ( 'Date Table'[Month] )
                || 'Primary Allocation Override'[Month] = "ALL" )
                && ( 'Primary Allocation Override'[Ship To Name] = SELECTEDVALUE ( 'Ship-To Master'[Ship To Name] )
                    || 'Primary Allocation Override'[Ship To Name] = "ALL" )
                && ( 'Primary Allocation Override'[Chain] = SELECTEDVALUE ( 'Chain Master'[Chain] )
                    || 'Primary Allocation Override'[Chain] = "ALL" )
                && ( 'Primary Allocation Override'[Brand] = SELECTEDVALUE ( 'Brand Master'[Brand] )
                    || 'Primary Allocation Override'[Brand] = "ALL" )
        )
    )
VAR _mapWeighted =
    DIVIDE (
        SUMX ( 'Primary Allocation Map', 'Primary Allocation Map'[Cont%] ),
        COUNTROWS ( 'Primary Allocation Map' )
    )
RETURN
    IF ( NOT ISBLANK ( _override ), DIVIDE ( _override, 100 ), _mapWeighted )

// ---------- (B) Compute allocation from an un-split raw primary ----------
// Point [Raw ShipTo Primary] at your un-split feed if/when one exists.
// By default it falls back to the already-allocated NSV so the measure is safe.
Raw ShipTo Primary =
COALESCE (
    // <-- replace with SUM('Raw Primary'[Primary NSV]) when an un-split feed is added
    BLANK (),
    [Ship-to Primary NSV]
)

// !! Do not place this on a report page while [Raw ShipTo Primary] still
// falls back to [Ship-to Primary NSV] (today's mode) -- see the
// DOUBLE-ALLOCATION LANDMINE note at the top of this file. Use
// [Ship-to Primary NSV] instead; that is what Page 2B actually uses today.
Allocated Primary (via Cont%) =
SUMX (
    SUMMARIZE (
        'Primary Allocation Map',
        'Primary Allocation Map'[Ship To Name],
        'Primary Allocation Map'[Chain],
        'Primary Allocation Map'[Brand]
    ),
    VAR _cont = [Primary Allocation Cont%]
    VAR _raw =
        CALCULATE (
            [Raw ShipTo Primary],
            ALL ( 'Chain Master'[Chain] )   // raw is at ship-to x brand (chain-agnostic)
        )
    RETURN _raw * _cont
)

// ---------- Secondary / Offtake at the same grain ----------
Secondary Offtake NSV = [Total Offtake NSV]
Secondary Offtake Qty = [Total Offtake Qty]

// ---------- Article inclusion: offtake this month OR previous month ----------
// Primary can lead/lag secondary by a month, so an article counts as "active"
// if it had offtake in the selected month or the one before.
Article Has Offtake (Curr or Prev) =
VAR _curr = [Total Offtake NSV]

// =====================================================================
// TOT% MEASURES  (Trade Offer Terms % / On-Invoice Margin Pass-on %)
//
// TOT% = SUM(Pass-on Value) / SUM(MRP) -- never a simple average of
// row-level TOT% percentages. Pass-on Value is sourced per row with a
// 3-TIER PRIORITY (mirrors dashboard/scripts/build_dashboard_data.py's
// tot_block() exactly, so a chain/category's TOT% reads the same on the
// HTML dashboard and here):
//
//   1. SOURCE      -- 'Fact Primary Article'[Avg TOT] (query 16 — the
//                     Primary file's own Customer x Article-grain TOT%, a
//                     0-1 fraction). Pass-on Value = Primary MRP x Avg TOT.
//   2. ACTUAL TAX  -- if Avg TOT is blank/invalid, use the row's actual
//                     [Primary Tax Amount] ("Inv. Tax Amount(LOC)"):
//                     Pass-on Value = MRP - NSV - Tax.

    )

//                       price / mix effect = units_now x change in ASP.  The two add up to the NSV change.
// =====================================================================

// ---------- INTERNAL MT OFFTAKE ----------
MT Offtake Qty = CALCULATE ( [Total Offtake Qty], 'Chain Master'[Channel] = "MT" )
MT MRP Sales   = CALCULATE ( [Total MRP Sales],   'Chain Master'[Channel] = "MT" )

MT ASP = DIVIDE ( [Total MT NSV], [MT Offtake Qty] )

// NSV as a share of MRP sales: how much of the shelf price we keep
MT Realisation % = DIVIDE ( [Total MT NSV], [MT MRP Sales] )

MT Offtake Qty PM = CALCULATE ( [MT Offtake Qty], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
MT ASP PM         = CALCULATE ( [MT ASP],         DATEADD ( 'Date Table'[Date], -1, MONTH ) )

MT Volume Effect MoM =
IF ( ISBLANK ( [MT ASP PM] ), BLANK (), [MT ASP PM] * ( [MT Offtake Qty] - [MT Offtake Qty PM] ) )

MT Price Mix Effect MoM =
IF ( ISBLANK ( [MT ASP PM] ), BLANK (), [MT Offtake Qty] * ( [MT ASP] - [MT ASP PM] ) )

MT Price Volume Check MoM =
[MT Volume Effect MoM] + [MT Price Mix Effect MoM] - ( [Total MT NSV] - CALCULATE ( [Total MT NSV], DATEADD ( 'Date Table'[Date], -1, MONTH ) ) )

MT Volume Effect YoY =
VAR _asp = CALCULATE ( [MT ASP], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
VAR _qty = CALCULATE ( [MT Offtake Qty], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
RETURN IF ( ISBLANK ( _asp ), BLANK (), _asp * ( [MT Offtake Qty] - _qty ) )

MT Price Mix Effect YoY =
VAR _asp = CALCULATE ( [MT ASP], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
RETURN IF ( ISBLANK ( _asp ), BLANK (), [MT Offtake Qty] * ( [MT ASP] - _asp ) )

// zone (or chain / brand) ASP against the MT average, 100 = average
Zone Price Index =
DIVIDE ( [MT ASP], CALCULATE ( [MT ASP], ALL ( 'Fact Offtake Sales'[Zone] ) ) ) * 100

Chain Price Index =
DIVIDE ( [MT ASP], CALCULATE ( [MT ASP], ALL ( 'Chain Master'[Chain] ) ) ) * 100

// ---------- NIELSEN (consumer offtake, IN URB MT) ----------
// Fact Nielsen[Volume 000 L] and [Price Per Ml] are loaded when the drop file carries them.
Nielsen Volume 000 L = SUM ( 'Fact Nielsen'[Volume 000 L] )

// Rs per ml = Rs / ( thousand litres x 1000 x 1000 )
Nielsen Price Per Ml = DIVIDE ( [Our Brand Sales], [Nielsen Volume 000 L] * 1000000 )

Nielsen Volume Effect YoY =
VAR _vol = CALCULATE ( [Nielsen Volume 000 L], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
VAR _ppml = CALCULATE ( [Nielsen Price Per Ml], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
RETURN IF ( ISBLANK ( _ppml ), BLANK (), ( [Nielsen Volume 000 L] - _vol ) * _ppml * 1000000 )

Nielsen Price Mix Effect YoY =
VAR _ppml = CALCULATE ( [Nielsen Price Per Ml], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
RETURN IF ( ISBLANK ( _ppml ), BLANK (), [Nielsen Volume 000 L] * ( [Nielsen Price Per Ml] - _ppml ) * 1000000 )

// ---------- CATEGORY PRESENCE (our own mix; Nielsen sizes only Facewash and Shampoo) ----------
// A sub-category that earns a real share of our NSV in all chains but little in one chain is a
// presence gap. Market size by category needs a Nielsen export for that category.
Subcat Share In Chain % =
DIVIDE ( [Total MT NSV], CALCULATE ( [Total MT NSV], ALL ( 'Fact Offtake Sales'[Sub-category] ) ) )

Subcat Share All Chains % =
DIVIDE (
    CALCULATE ( [Total MT NSV], ALL ( 'Chain Master'[Chain] ) ),
    CALCULATE ( [Total MT NSV], ALL ( 'Chain Master'[Chain] ), ALL ( 'Fact Offtake Sales'[Sub-category] ) )
)

Subcat Presence Status =
VAR _all = [Subcat Share All Chains %]
VAR _in = COALESCE ( [Subcat Share In Chain %], 0 )
RETURN
    IF ( ISBLANK ( _all ) || _all < 0.02, BLANK (),
        IF ( _in <= 0, "Absent", IF ( _in < _all * 0.6, "Under-weight", "Present" ) ) )

Subcat Gap NSV =
VAR _gap = MAX ( [Subcat Share All Chains %] - COALESCE ( [Subcat Share In Chain %], 0 ), 0 )
RETURN IF ( [Subcat Presence Status] IN { "Absent", "Under-weight" },
    _gap * CALCULATE ( [Total MT NSV], ALL ( 'Fact Offtake Sales'[Sub-category] ) ) )

// ---------- CORPORATE-VISIT CITIES (tables: 'Store City Master', 'Visit City List') ----------
// Needs 'Fact Offtake Sales'[Match Key] (calculated column, see 49_Store_City_Master.pq) related to 'Store City Master'[Match Key].
Visit Stores = DISTINCTCOUNT ( 'Store City Master'[Store Key] )

Visit City NSV =
CALCULATE ( [Total MT NSV], 'Store City Master'[Visit Status] = "Considered" )

Near Listed NSV =
CALCULATE ( [Total MT NSV], 'Store City Master'[Visit Status] = "Near listed city" )

// share of MT NSV that is attached to a store in the master (should be close to 100%)
Visit Master Match % =
DIVIDE (
    CALCULATE ( [Total MT NSV], NOT ISBLANK ( 'Store City Master'[Store Key] ) ),
    [Total MT NSV]
)

Visit City NSV MoM % =
VAR _pm = CALCULATE ( [Visit City NSV], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
RETURN DIVIDE ( [Visit City NSV] - _pm, _pm )

// ---------- ONE STORE UNIVERSE (Executive Cockpit and Inventory & Supply Health use the same count) ----------
// POS stores = distinct Store Code selling in the selected month(s). Chains in 'Store Universe Overrides' are taken from that table
// instead: TENTATIVE counts are added, NA chains (Nykaa / FSN) are not counted.
Universe POS Stores =
CALCULATE (
    DISTINCTCOUNT ( 'Fact Offtake Sales'[Store Code] ),
    NOT ( 'Fact Offtake Sales'[Chain] IN VALUES ( 'Store Universe Overrides'[Offtake Chain] ) )
)

Universe Tentative Stores =
CALCULATE ( SUM ( 'Store Universe Overrides'[Store Count] ), 'Store Universe Overrides'[Status] = "TENTATIVE" )

Universe Stores = [Universe POS Stores] + COALESCE ( [Universe Tentative Stores], 0 )

Universe Stores Note =
"Includes ~" & FORMAT ( COALESCE ( [Universe Tentative Stores], 0 ), "#,0" ) & " tentative stores (Reliance Retail, More Retail); Nykaa / FSN is NA"

// actual universe (stores in the maintained store master, Brand Counter counters left out) against the stores counted as available
Universe Master Stores =
CALCULATE (
    DISTINCTCOUNT ( 'Store City Master'[Store Key] ),
    NOT ( CONTAINSSTRING ( 'Store City Master'[Store Key], "|NO-CITY" ) ),
    'Store City Master'[Chain Name] <> "Reliance Brand Counter"
)

Universe Availability % = DIVIDE ( [Universe Stores], [Universe Master Stores] )

// =====================================================================
// ACCOUNT (RETAILER) CATEGORY SHARE  (tables: 'Fact Account Category', 'Fact Account Category Geo', 'Fact Account Assortment')
// Account sales = the chain's own category sales; our sales = Honasa sales in the same categories, Rs lakh without GST.
// Reliance Retail and Reliance Brand Counter (the staffed counters) come from the Reliance file: gross sales as supplied, so compare shares, not rupees.
// Never add Reliance Retail and Reliance Brand Counter together; they are two separate sets of stores.
// Lulu and More Retail cover the categories and stores where we sell; More Retail months before Jun 26 are on a smaller scope:
// read More Retail month on month only within Jun-Aug 26 (the Scope column says which scope a row is on).
// For "Level" use "Category" (class level for More Retail) unless you want More Retail sub-classes.
// =====================================================================
Account Sales Lakh = CALCULATE ( SUM ( 'Fact Account Category'[Account Sales Rs L] ), 'Fact Account Category'[Level] = "Category" )
Our Account Sales Lakh = CALCULATE ( SUM ( 'Fact Account Category'[Honasa Sales Rs L] ), 'Fact Account Category'[Level] = "Category" )

Account Share % = DIVIDE ( [Our Account Sales Lakh], [Account Sales Lakh] )

Account Sales MoM % =
VAR _pm = CALCULATE ( [Account Sales Lakh], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
RETURN DIVIDE ( [Account Sales Lakh] - _pm, _pm )

Our Account Sales MoM % =
VAR _pm = CALCULATE ( [Our Account Sales Lakh], DATEADD ( 'Date Table'[Date], -1, MONTH ) )
RETURN DIVIDE ( [Our Account Sales Lakh] - _pm, _pm )

Account Share Change pp =
( [Account Share %] - CALCULATE ( [Account Share %], DATEADD ( 'Date Table'[Date], -1, MONTH ) ) ) * 100

Account Face Wash Share % =
CALCULATE ( [Account Share %], 'Fact Account Category'[Common Category] = "Face Wash & Cleanser" )

// the category's weight in the chain's category sales, and our chain-wide share (all categories in the same chain and month)
Account Category % Of Chain =
DIVIDE ( [Account Sales Lakh], CALCULATE ( [Account Sales Lakh], ALL ( 'Fact Account Category'[Category], 'Fact Account Category'[Common Category] ) ) )

Account Chain Share % =
CALCULATE ( [Account Share %], ALL ( 'Fact Account Category'[Category], 'Fact Account Category'[Common Category] ) )

Our Articles In Category = MAX ( 'Fact Account Assortment'[Articles] )

// white space: category is 2% or more of the chain and our share is under 0.5%; low assortment: 3 or fewer articles and below the chain share
// only categories that matter to us (at least 0.5% of our own sales across the chains, not "Other") are ever flagged: [Relevant For Us] = "No" gives a blank
Account White Space Flag =
VAR _rel = SELECTEDVALUE ( 'Fact Account Category'[Relevant For Us], "Yes" )
VAR _pct = [Account Category % Of Chain]
VAR _share = COALESCE ( [Account Share %], 0 )
VAR _chain = [Account Chain Share %]
VAR _art = [Our Articles In Category]
RETURN
    IF ( _rel = "No" || ISBLANK ( _pct ) || _pct < 0.02, BLANK (),
        IF ( _share < 0.005, "White space",
            IF ( NOT ISBLANK ( _art ) && _art <= 3 && _share < _chain, "Low assortment",
                IF ( _share < _chain * 0.5, "Below chain average",
                    IF ( _share >= _chain * 1.5, "Strong", BLANK () ) ) ) ) )

// our share in the rest of our range (without Face Wash and Shampoo): the realistic target for a thin category
Account Rest Of Range Share % =
CALCULATE ( [Account Share %], ALL ( 'Fact Account Category'[Category], 'Fact Account Category'[Common Category] ),
    NOT ( 'Fact Account Category'[Common Category] IN { "Face Wash & Cleanser", "Shampoo" } ) )

// gap to that target, Rs lakh a month, for white space / low assortment / below average rows (Face Wash and Shampoo use the chain-wide share)
Account Share Gap Lakh =
VAR _c = SELECTEDVALUE ( 'Fact Account Category'[Common Category] )
VAR _target = IF ( _c IN { "Face Wash & Cleanser", "Shampoo" }, [Account Chain Share %], [Account Rest Of Range Share %] )
RETURN
IF ( [Account White Space Flag] IN { "White space", "Low assortment", "Below chain average" },
    MAX ( _target - COALESCE ( [Account Share %], 0 ), 0 ) * [Account Sales Lakh] )

Account Scope Note = SELECTEDVALUE ( 'Fact Account Category'[Scope], "More than one scope selected" )

// zone / state / city cuts (Lulu, More Retail and Reliance; Wellness Forever has category level only)
Account Geo Sales Lakh = SUM ( 'Fact Account Category Geo'[Account Sales Rs L] )
Our Geo Sales Lakh = SUM ( 'Fact Account Category Geo'[Honasa Sales Rs L] )
Account Geo Share % = DIVIDE ( [Our Geo Sales Lakh], [Account Geo Sales Lakh] )

// =====================================================================
// NIELSEN BRAND CUTS  (table: 'Fact Nielsen Brand Cut'; the Nielsen Cuts tab of the market-share report)
// Use with Nielsen Category and Brand slicers and one month. WD, ND and SAH are percentage points; shares are decimals.
// The "(Category)" row (Is Category Row = "Yes") holds the category value and price per ml; brand visuals should filter it out.

Cut Brand Value Cr = CALCULATE ( SUM ( 'Fact Nielsen Brand Cut'[Value Cr] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut Category Value Cr = CALCULATE ( SUM ( 'Fact Nielsen Brand Cut'[Value Cr] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "Yes", REMOVEFILTERS ( 'Fact Nielsen Brand Cut'[Brand] ) )
Cut Value Share % = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[Value Share %] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut Volume Share % = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[Volume Share %] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
// value share above volume share = the brand sells at a price above the category average
Cut Value Less Volume Share pp = ( [Cut Value Share %] - [Cut Volume Share %] ) * 100

Cut Brand Price Per Ml = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[Price Per Ml] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut Category Price Per Ml =
CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[Price Per Ml] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "Yes", REMOVEFILTERS ( 'Fact Nielsen Brand Cut'[Brand] ) )
// 100 = the category price per ml
Cut Price Index = DIVIDE ( [Cut Brand Price Per Ml], [Cut Category Price Per Ml] ) * 100
Cut Price Per Ml YoY % =
VAR _ly = CALCULATE ( [Cut Brand Price Per Ml], DATEADD ( 'Date Table'[Date], -12, MONTH ) )
RETURN DIVIDE ( [Cut Brand Price Per Ml] - _ly, _ly )

Cut WD % = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[WD %] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut ND % = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[ND %] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut WD Change pp = [Cut WD %] - CALCULATE ( [Cut WD %], DATEADD ( 'Date Table'[Date], -12, MONTH ) )

Cut WD Less ND pp = [Cut WD %] - [Cut ND %]
Cut Stores = CALCULATE ( SUM ( 'Fact Nielsen Brand Cut'[Stores] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut Sales Per Store Rs = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[Sales Per Store Rs] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )
Cut SAH % = CALCULATE ( AVERAGE ( 'Fact Nielsen Brand Cut'[SAH %] ), 'Fact Nielsen Brand Cut'[Is Category Row] = "No" )

Cut Share Per WD Point = DIVIDE ( [Cut Value Share %] * 100, [Cut WD %] )

Cut Share At 95 WD % = IF ( [Cut WD %] < 95, [Cut SAH %] / 100 * 0.95 )
Cut Share Headroom pp = IF ( ISBLANK ( [Cut Share At 95 WD %] ), BLANK (), ( [Cut Share At 95 WD %] - [Cut Value Share %] ) * 100 )
// rupee value of one share point in this category, Rs Cr a month
Cut Value Of One Share Point Cr = DIVIDE ( [Cut Category Value Cr], 100 )
